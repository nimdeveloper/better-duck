1//! Maintainer tooling for the `better-duck` workspace. Not published.
2//!
3//! `cargo run -p xtask -- upgrade-duckdb --tag v1.5.5` regenerates
4//! `crates/better-duck-sys/vendor/duckdb.tar.gz` and
5//! `crates/better-duck-sys/src/bindings.rs` from a specific upstream
6//! `duckdb/duckdb` release tag — the *only* place a git checkout of DuckDB's
7//! source ever touches this repo. The clone is ephemeral (a `tempfile`
8//! directory, deleted when this process exits, on any exit path); nothing is
9//! left behind as a submodule reference, and no consumer of
10//! `better-duck-sys` ever needs network access or Python to build it.
11//!
12//! This calls into DuckDB's own `scripts/package_build.py`
13//! (`build_package(...)`) to get the amalgamated (unity-build) source file
14//! list — the same officially-provided mechanism the community
15//! `libduckdb-sys` crate's own `update_sources.py` uses — rather than
16//! reimplementing DuckDB's internal source-selection logic in Rust, which
17//! would be fragile against upstream changes.

47/// This crate's own manifest schema, written into the vendored archive for
48/// `better-duck-sys/build.rs` to consume. Deliberately simpler than DuckDB's
49/// own `package_build.py` output — we only need "which files, which include
50/// dirs, plus one optional extra source list per Cargo feature".
51#[derive(Serialize, Deserialize)]
52struct Manifest {
53    duckdb_version: String,
54    sources: Vec<String>,
55    json_sources: Vec<String>,
56    parquet_sources: Vec<String>,
57    include_dirs: Vec<String>,
58}
59
60fn main() -> Result<()> {
61    let cli = Cli::parse();
62    match cli.command {
63        Command_::UpgradeDuckdb { tag } => upgrade_duckdb(&tag),
64    }
65}

67fn workspace_root() -> Result<PathBuf> {
68    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
69    manifest_dir.parent().map(Path::to_path_buf).context(
70        "xtask's CARGO_MANIFEST_DIR has no parent — expected xtask/ under the workspace root",
71    )
72}

74fn upgrade_duckdb(tag: &str) -> Result<()> {
75    let root = workspace_root()?;
76    let sys_crate = root.join("crates").join("better-duck-sys");
77    if !sys_crate.is_dir() {
78        bail!("expected {} to exist", sys_crate.display());
79    }
80
81    let work = tempfile_dir("better-duck-xtask-duckdb")?;
82    println!("==> shallow-cloning duckdb/duckdb@{tag} into {}", work.display());
83    clone_tag(tag, &work)?;
84
85    // Deliberately NOT nested under `work` (the checkout): `package_build.py`
86    // returns some paths (from `amalgamation.list_sources()`) relative to the
87    // checkout root, so a staging dir inside the checkout would make
88    // `collect_sources`'s prefix-stripping ambiguous between "relative to the
89    // checkout" and "relative to staging".
90    let staging = tempfile_dir("better-duck-xtask-staging")?;
91    println!("==> collecting the amalgamated source file list via package_build.py");
92    let manifest = collect_sources(&work, &staging, tag)?;
93
94    println!("==> writing manifest.json + packaging vendor/duckdb.tar.gz");
95    let manifest_json = serde_json::to_string_pretty(&manifest)?;
96    fs::write(staging.join("manifest.json"), &manifest_json)?;
97    let vendor_dir = sys_crate.join("vendor");
98    fs::create_dir_all(&vendor_dir)?;
99    package_archive(&staging, &vendor_dir.join("duckdb.tar.gz"))?;
100
101    println!("==> regenerating src/bindings.rs via bindgen");
102    let header = work.join("src").join("include").join("duckdb.h");
103    if !header.is_file() {
104        bail!("expected DuckDB's C header at {}", header.display());
105    }
106    generate_bindings(&header, &sys_crate.join("src").join("bindings.rs"))?;
107
108    println!(
109        "==> done. Review the diff, then commit crates/better-duck-sys/{{vendor/duckdb.tar.gz,src/bindings.rs}}."
110    );
111    // `work` (the clone) is a tempfile::TempDir — deleted automatically when
112    // it drops here, on every return path including `?` early-returns above,
113    // since it's owned by this function's stack frame throughout.
114    Ok(())
115}

117/// A directory deleted on drop, regardless of how this function returns.
118/// Avoids taking a `tempfile` dependency for one call site: `TempDirGuard`
119/// wraps a plain path and removes it in `Drop`.
120struct TempDirGuard(PathBuf);
121impl Drop for TempDirGuard {
122    fn drop(&mut self) {
123        let _ = fs::remove_dir_all(&self.0);
124    }
125}
126impl std::ops::Deref for TempDirGuard {
127    type Target = Path;
128    fn deref(&self) -> &Path {
129        &self.0
130    }
131}
132
133fn tempfile_dir(prefix: &str) -> Result<TempDirGuard> {
134    let base = std::env::temp_dir();
135    let unique = format!("{prefix}-{}", std::process::id());
136    let dir = base.join(unique);
137    if dir.exists() {
138        fs::remove_dir_all(&dir)?;
139    }
140    fs::create_dir_all(&dir)?;
141    Ok(TempDirGuard(dir))
142}

144fn clone_tag(
145    tag: &str,
146    dest: &Path,
147) -> Result<()> {
148    let status = Command::new("git")
149        .args(["clone", "--depth", "1", "--branch", tag, "https://github.com/duckdb/duckdb.git"])
150        .arg(dest)
151        .status()
152        .context("failed to run `git clone` — is git installed and on PATH?")?;
153    if !status.success() {
154        bail!("git clone of duckdb/duckdb@{tag} failed (exit code {status})");
155    }
156    Ok(())
157}

159/// Runs a small embedded Python helper (mirroring the relevant part of
160/// `libduckdb-sys`'s own `update_sources.py`) that imports DuckDB's
161/// `scripts/package_build.py` and calls `build_package(...)` to get the
162/// amalgamated source/include lists, then copies exactly those files into
163/// `staging`.
164fn collect_sources(
165    duckdb_checkout: &Path,
166    staging: &Path,
167    tag: &str,
168) -> Result<Manifest> {
169    fs::create_dir_all(staging)?;
170    let helper = duckdb_checkout.join("_better_duck_collect_sources.py");
171    fs::write(&helper, COLLECT_SOURCES_PY)?;
172
173    // Outside `staging` deliberately: everything under `staging` gets tar'd up
174    // verbatim as the vendored archive (see `package_archive`), and this file
175    // is just an intermediate handoff to the Rust side, not part of the manifest.
176    let output_json = duckdb_checkout.join("_raw_sources.json");
177    let status = Command::new("python3")
178        .arg(&helper)
179        .arg(duckdb_checkout)
180        .arg(staging)
181        .arg(&output_json)
182        .status()
183        .context("failed to run python3 — is Python 3 installed and on PATH?")?;
184    if !status.success() {
185        bail!("package_build.py source collection failed (exit code {status})");
186    }
187
188    #[derive(Deserialize)]
189    struct RawSources {
190        base_cpp_files: Vec<String>,
191        base_include_dirs: Vec<String>,
192        json_cpp_files: Vec<String>,
193        parquet_cpp_files: Vec<String>,
194    }
195    let raw: RawSources = serde_json::from_str(
196        &fs::read_to_string(&output_json).context("reading intermediate sources JSON")?,
197    )?;
198
199    Ok(Manifest {
200        duckdb_version: tag.to_owned(),
201        sources: raw.base_cpp_files,
202        json_sources: raw.json_cpp_files,
203        parquet_sources: raw.parquet_cpp_files,
204        include_dirs: raw.base_include_dirs,
205    })
206}

208/// Copies `staging`'s contents into a `tar.gz` at `dest`, sorted, with a
209/// fixed mtime — reproducible archive bytes for a given source tree, so an
210/// unrelated re-run doesn't churn the committed archive.
211fn package_archive(
212    staging: &Path,
213    dest: &Path,
214) -> Result<()> {
215    let file = fs::File::create(dest)?;
216    let encoder = flate2::write::GzEncoder::new(file, flate2::Compression::best());
217    let mut builder = tar::Builder::new(encoder);
218    let mut entries: Vec<PathBuf> = walkdir::WalkDir::new(staging)
219        .into_iter()
220        .filter_map(|e| e.ok())
221        .filter(|e| e.file_type().is_file())
222        .map(|e| e.path().to_path_buf())
223        .collect();
224    entries.sort();
225    for path in entries {
226        let rel = path.strip_prefix(staging)?;
227        builder.append_path_with_name(&path, rel)?;
228    }
229    builder.into_inner()?.finish()?;
230    Ok(())
231}

233fn generate_bindings(
234    header: &Path,
235    dest: &Path,
236) -> Result<()> {
237    let bindings = bindgen::Builder::default()
238        .generate_comments(true)
239        .raw_line("#![allow(rustdoc::invalid_html_tags)]")
240        .raw_line("#![allow(rustdoc::broken_intra_doc_links)]")
241        .raw_line("#![allow(rustdoc::invalid_rust_codeblocks)]")
242        .raw_line("#![allow(rustdoc::bare_urls)]")
243        .header(header.to_string_lossy().to_string())
244        .allowlist_function("duckdb_.*")
245        .allowlist_type("duckdb_.*")
246        .allowlist_var("DUCKDB_.*")
247        .generate()
248        .map_err(|e| anyhow::anyhow!("bindgen failed: {e}"))?;
249    bindings.write_to_file(dest).context("writing generated bindings")?;
250    Ok(())
251}

253/// Embedded helper — kept as an external-process Python script (not
254/// reimplemented in Rust) so it can call DuckDB's own
255/// `scripts/package_build.py` directly, exactly like `libduckdb-sys`'s own
256/// `update_sources.py` does. Only used by this xtask, at maintainer-upgrade
257/// time; never runs on a consumer's machine.
258const COLLECT_SOURCES_PY: &str = r#"
259import json
260import shutil
261import sys
262from pathlib import Path
263
264duckdb_checkout = Path(sys.argv[1]).resolve()
265staging = Path(sys.argv[2]).resolve()
266output_json = Path(sys.argv[3]).resolve()
267
268# `package_build.build_package(target_dir, ...)` (default `folder_name="duckdb"`)
269# returns two different path shapes in its source list: ordinary (non-unity-
270# build) files come back as `"duckdb/<relative-to-checkout>"` — meant to be
271# resolved against `target_dir`'s *parent* — while unity-build (`ub_*.cpp`)
272# files it generates itself come back as absolute paths directly under
273# `target_dir`. Passing `target_dir = staging/"duckdb"` makes both shapes
274# consistent once normalized below: everything ends up expressed relative to
275# `staging`, matching where the files were actually written on disk (which is
276# what gets tar'd up as the vendored archive).
277target_dir = staging / "duckdb"
278staging_prefix = staging.as_posix() + "/"
279
280sys.path.append(str(duckdb_checkout / "scripts"))
281import package_build
282
283def normalize(path_str):
284    p = path_str.replace("\\", "/")
285    if p.startswith(staging_prefix):
286        return p[len(staging_prefix):]
287    return p  # already "duckdb/<relative>" per folder_name
288
289# `core_functions` (basic scalar/aggregate functions like `sum`, `abs`, ...)
290# is unconditionally required — DuckDB's own `extension/extension_config.cmake`
291# calls it out by name as "loaded by default on every build as [it is] an
292# essential part of DuckDB", unlike `json`/`parquet` which really are optional
293# file-format support. It's always in `default_linked_extensions` below so its
294# `DUCKDB_EXTENSION_CORE_FUNCTIONS_LINKED` fallback in the generated loader is
295# unconditionally 1 — no Cargo feature gates it.
296extensions = ["core_functions", "json", "parquet"]
297# `default_linked_extensions` matters: `build_package` bakes each extension's
298# "is it linked" flag into the *content* of the single shared
299# `generated_extension_loader_package_build.cpp` file as
300# `#ifndef DUCKDB_EXTENSION_X_LINKED / #define ... <default> / #endif`. For
301# `json`/`parquet` that fallback is 0 (off), and the `#ifndef` guard means our
302# own `-D DUCKDB_EXTENSION_X_LINKED` (added by `better-duck-sys/build.rs` only
303# when the matching Cargo feature is on) wins whenever it's present. This file
304# is compiled unconditionally as a "base" source, so json/parquet's behavior
305# must be controlled entirely by our own `-D` flags, not by whatever extension
306# set happened to be passed to `build_package` here.
307(source_list, include_list, _) = package_build.build_package(
308    str(target_dir), extensions, False, default_linked_extensions=["core_functions"]
309)
310loader = duckdb_checkout / "generated_extension_loader_package_build.cpp"
311loader.unlink(missing_ok=True)
312
313sources = {normalize(s) for s in source_list}
314# `include_list` isn't run through the `folder_name` convention by
315# `build_package` itself (unlike `source_list`) even though the actual
316# header files live under `target_dir` just like the sources — prefix it
317# the same way by hand so `-I` dirs resolve consistently with `sources`.
318includes = {f"duckdb/{i}" for i in include_list}
319
320# A single `build_package` call (rather than one call per extension
321# combination) avoids each call clobbering the previous one's copy of the
322# shared loader file on disk. Split the one unified list back apart by each
323# extension's own source directory instead.
324json_sources = {s for s in sources if "duckdb/extension/json/" in s}
325parquet_sources = {s for s in sources if "duckdb/extension/parquet/" in s}
326base_sources = sources - json_sources - parquet_sources
327
328result = {
329    "base_cpp_files": sorted(base_sources),
330    "base_include_dirs": sorted(includes),
331    "json_cpp_files": sorted(json_sources),
332    "parquet_cpp_files": sorted(parquet_sources),
333}
334with output_json.open("w") as f:
335    json.dump(result, f, indent=2, sort_keys=True)
336"#;
