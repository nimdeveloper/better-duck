43/// Implementation of `FromSql` for bool values.
44impl FromSql<Bool, DuckDb> for bool {
45    fn from_sql(val: DuckValueRef<'_>) -> deserialize::Result<Self> {
46        match val {
47            DuckValueRef::Boolean(v) => Ok(v),
48            _ => Err("Unexpected data for boolean type".into()),
49        }
50    }
51}
52
53/// Implementation of `ToSql` for bool values.
54impl ToSql<Bool, DuckDb> for bool {
55    fn to_sql<'b>(
56        &'b self,
57        out: &mut Output<'b, '_, DuckDb>,
58    ) -> serialize::Result {
59        out.set_value(DuckValueRef::Boolean(*self));
60        Ok(IsNull::No)
61    }
62}
