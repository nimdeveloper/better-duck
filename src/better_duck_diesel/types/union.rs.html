1//! `FromSql`/`ToSql` implementations for DuckDB `UNION` / Diesel `DuckUnion`.
2//!
3//! The Rust mirror type is `Box<DuckValue>` — the active member's value — matching
4//! `better_duck_core::types::value::DuckValue::Union` exactly. The member name and
5//! tag index are not currently surfaced (see the core-level roadmap for multi-arm
6//! union support).
7
8use better_duck_core::types::{value::DuckValue, value_ref::DuckValueRef};
9use diesel::{
10    deserialize::{self, FromSql},
11    serialize::{self, IsNull, Output, ToSql},
12};
13
14use crate::backend::DuckDb;
15use crate::types::duckdb_types::DuckUnion;
16
17// FromSql
18
19/// Deserialize a DuckDB `UNION` column into a [`Box<DuckValue>`] holding the
20/// active member's value.
21impl FromSql<DuckUnion, DuckDb> for Box<DuckValue> {
22    fn from_sql(val: DuckValueRef<'_>) -> deserialize::Result<Self> {
23        match val {
24            DuckValueRef::Union(inner) => Ok(Box::new(DuckValue::from(inner.as_ref()))),
25            other => Err(format!("expected Union, got {other:?}").into()),
26        }
27    }
28}
29
30// ToSql
31
32/// Serialize a [`Box<DuckValue>`] as a DuckDB `UNION` bind parameter.
33impl ToSql<DuckUnion, DuckDb> for Box<DuckValue> {
34    fn to_sql<'b>(
35        &'b self,
36        out: &mut Output<'b, '_, DuckDb>,
37    ) -> serialize::Result {
38        let inner = DuckValueRef::from((**self).clone());
39        out.set_value(DuckValueRef::Union(Box::new(inner)));
40        Ok(IsNull::No)
41    }
42}
