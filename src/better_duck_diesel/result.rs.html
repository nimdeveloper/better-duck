1//! Error conversion between `better_duck_core` errors and Diesel's error types.
2
3use std::fmt;
4
5/// Wraps a [`better_duck_core::error::Error`] for conversion into a Diesel error.
6pub struct DuckDbError {
7    orig: better_duck_core::error::Error,
8}
9
10impl DuckDbError {
11    /// Creates a new [`DuckDbError`] wrapping a core error.
12    pub fn new(e: better_duck_core::error::Error) -> Self {
13        DuckDbError { orig: e }
14    }
15}
16
17impl fmt::Display for DuckDbError {
18    fn fmt(
19        &self,
20        f: &mut fmt::Formatter<'_>,
21    ) -> fmt::Result {
22        self.orig.fmt(f)
23    }
24}
25
26impl fmt::Debug for DuckDbError {
27    fn fmt(
28        &self,
29        f: &mut fmt::Formatter<'_>,
30    ) -> fmt::Result {
31        write!(f, "DuckDbError({:?})", self.orig)
32    }
33}
34
35impl From<DuckDbError> for diesel::result::Error {
36    fn from(e: DuckDbError) -> Self {
37        use better_duck_core::error::Error as CE;
38        use diesel::result::{DatabaseErrorKind as K, Error as DE};
39
40        match e.orig {
41            CE::QueryReturnedNoRows => DE::NotFound,
42            CE::NulError(n) => DE::InvalidCString(n),
43            CE::Utf8Error(u) => DE::SerializationError(Box::new(u)),
44            CE::ToSqlConversionFailure(b) => DE::SerializationError(b),
45            CE::InvalidColumnIndex(i) => {
46                DE::DeserializationError(format!("invalid column index {i}").into())
47            },
48            CE::InvalidColumnName(n) => {
49                DE::DeserializationError(format!("unknown column '{n}'").into())
50            },
51            CE::DuckDBFailure(_, msg) => {
52                let msg = msg.unwrap_or_else(|| "duckdb error".to_owned());
53                DE::DatabaseError(K::Unknown, Box::new(msg))
54            },
55            other => DE::DatabaseError(K::Unknown, Box::new(format!("{other}"))),
56        }
57    }
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63    use better_duck_core::{
64        error::Error as CoreError,
65        ffi::{DuckDBError, Error as FfiError},
66    };
67    use diesel::result::{DatabaseErrorKind, Error as DieselError};
68    use std::{ffi::CString, str};
69
70    fn utf8_error() -> str::Utf8Error {
71        let bytes = vec![0x80];
72        str::from_utf8(&bytes).expect_err("test input must be invalid UTF-8")
73    }
74
75    fn message(error: DieselError) -> String {
76        match error {
77            DieselError::DatabaseError(DatabaseErrorKind::Unknown, info) => info.message().into(),
78            other => panic!("expected unknown database error, got {other:?}"),
79        }
80    }
81
82    #[test]
83    fn display_and_debug_delegate_to_the_core_error() {
84        let wrapped = DuckDbError::new(CoreError::InvalidColumnIndex(7));
85        assert_eq!(wrapped.to_string(), "Invalid column index: 7");
86        assert!(format!("{wrapped:?}").contains("InvalidColumnIndex(7)"));
87    }
88
89    #[test]
90    fn maps_not_found_and_invalid_c_string() {
91        assert!(matches!(
92            DieselError::from(DuckDbError::new(CoreError::QueryReturnedNoRows)),
93            DieselError::NotFound
94        ));
95
96        let nul = CString::new(b"a\0b".to_vec()).unwrap_err();
97        assert!(matches!(
98            DieselError::from(DuckDbError::new(CoreError::NulError(nul))),
99            DieselError::InvalidCString(error) if error.nul_position() == 1
100        ));
101    }
102
103    #[test]
104    fn maps_serialization_errors() {
105        assert!(matches!(
106            DieselError::from(DuckDbError::new(CoreError::Utf8Error(utf8_error()))),
107            DieselError::SerializationError(error) if error.to_string().contains("utf-8")
108        ));
109
110        let source = std::io::Error::other("cannot serialize");
111        assert!(matches!(
112            DieselError::from(DuckDbError::new(CoreError::ToSqlConversionFailure(Box::new(source)))),
113            DieselError::SerializationError(error) if error.to_string() == "cannot serialize"
114        ));
115    }
116
117    #[test]
118    fn maps_column_errors_to_deserialization_errors() {
119        assert!(matches!(
120            DieselError::from(DuckDbError::new(CoreError::InvalidColumnIndex(4))),
121            DieselError::DeserializationError(error) if error.to_string() == "invalid column index 4"
122        ));
123        assert!(matches!(
124            DieselError::from(DuckDbError::new(CoreError::InvalidColumnName("total".into()))),
125            DieselError::DeserializationError(error) if error.to_string() == "unknown column 'total'"
126        ));
127    }
128
129    #[test]
130    fn maps_duckdb_failures_with_and_without_context() {
131        let with_context =
132            CoreError::DuckDBFailure(FfiError::new(DuckDBError), Some("constraint failed".into()));
133        assert_eq!(message(DuckDbError::new(with_context).into()), "constraint failed");
134
135        let without_context = CoreError::DuckDBFailure(FfiError::new(DuckDBError), None);
136        assert_eq!(message(DuckDbError::new(without_context).into()), "duckdb error");
137    }
138
139    #[test]
140    fn maps_other_core_errors_to_unknown_database_errors() {
141        assert_eq!(
142            message(DuckDbError::new(CoreError::InvalidParameterCount(1, 2)).into()),
143            "Wrong number of parameters passed to query. Got 1, needed 2"
144        );
145    }
