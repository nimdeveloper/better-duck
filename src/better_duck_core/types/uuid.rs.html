1//! [`DuckDialect`] implementation for DuckDB's `UUID` type.
2//!
3//! UUID's physical column storage is fixed-width (16 bytes, like HUGEINT), so reads
4//! from a data chunk vector use the packed `duckdb_uhugeint` struct directly (via
5//! `read_packed!` in `value.rs`) rather than going through a heap-allocated
6//! `duckdb_value`.
7
8use super::*;
9use crate::types::appendable::AppendAble;
10use crate::{
11    ffi::{
12        duckdb_create_logical_type, duckdb_create_uuid, duckdb_logical_type, duckdb_uhugeint,
13        duckdb_value, DUCKDB_TYPE_DUCKDB_TYPE_UUID,
14    },
15    impl_appendable_via_to_duck_native,
16};
17
18/// A DuckDB `UUID` value: a 128-bit value in standard big-endian bit order (the
19/// order you'd get from `u128::from_be_bytes` on the 16-byte UUID representation).
20///
21/// DuckDB stores UUIDs internally as a `duckdb_uhugeint` with the most-significant
22/// bit of the upper half flipped, so that unsigned hugeint comparisons sort the
23/// same way as byte-wise UUID comparisons. `duckdb_create_uuid` performs that flip
24/// itself, so [`to_duck`](DuckUuid::to_duck) passes the value straight through;
25/// [`from_duck`](DuckUuid::from_duck) reads the already-flipped physical storage
26/// directly (bypassing the value API), so it must undo the flip itself. This type
27/// hides all of that: `DuckUuid(0)` is the nil UUID, and ordering on `DuckUuid`
28/// matches standard UUID ordering.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
30pub struct DuckUuid(pub u128);
31
32const UUID_SIGN_FLIP: u64 = 0x8000_0000_0000_0000;
33
34impl DuckUuid {
35    /// Creates a `DuckUuid` from its 128-bit value in standard big-endian bit order.
36    #[inline]
37    pub fn new(value: u128) -> DuckUuid {
38        DuckUuid(value)
39    }
40}
41
42impl DuckDialect<duckdb_uhugeint> for DuckUuid {
43    fn from_duck(raw: duckdb_uhugeint) -> Result<Self, DuckDBConversionError> {
44        // Undo the sign-bit flip `duckdb_create_uuid` applies internally — this read
45        // path pulls the raw physical struct directly out of the chunk vector, bypassing
46        // the value API that would otherwise undo it for us.
47        let upper = raw.upper ^ UUID_SIGN_FLIP;
48        Ok(DuckUuid((upper as u128) << 64 | (raw.lower as u128)))
49    }
50
51    fn to_duck(&self) -> Result<duckdb_value, DuckDBConversionError> {
52        let upper = (self.0 >> 64) as u64;
53        let lower = self.0 as u64;
54        // SAFETY: `duckdb_uhugeint { lower, upper }` is a valid FFI value for any bit
55        // pattern. `duckdb_create_uuid` performs the sign-bit flip itself, so `upper` is
56        // passed through unflipped here.
57        Ok(unsafe { duckdb_create_uuid(duckdb_uhugeint { lower, upper }) })
58    }
59}
60
61impl_appendable_via_to_duck_native!(DuckUuid);
62
63impl DuckLogicalType for DuckUuid {
64    fn duck_logical_type() -> Result<duckdb_logical_type, DuckDBConversionError> {
65        // SAFETY: DUCKDB_TYPE_DUCKDB_TYPE_UUID is always a valid duckdb_type constant.
66        Ok(unsafe { duckdb_create_logical_type(DUCKDB_TYPE_DUCKDB_TYPE_UUID) })
67    }
68}
69
70impl From<DuckUuid> for value::DuckValue {
71    fn from(u: DuckUuid) -> Self {
72        value::DuckValue::Uuid(u)
73    }
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79    use crate::ffi::duckdb_destroy_value;
80
81    /// `duckdb_get_uuid`/`duckdb_create_uuid` operate on a bare `duckdb_value` and do
82    /// not apply the storage-level sign flip — that flip only happens once a value is
83    /// actually written into a column's physical vector (verified empirically via the
84    /// end-to-end `rt_uuid_*` tests in `tests/types_roundtrip.rs`, which go through a
85    /// real INSERT + SELECT). So `to_duck` must be a plain passthrough, checked here.
86    fn to_duck_is_passthrough(value: DuckUuid) {
87        use crate::ffi::duckdb_get_uuid;
88        let mut duck_value = value.to_duck().unwrap();
89        // SAFETY: `duck_value` was created by `to_duck` above.
90        let raw = unsafe { duckdb_get_uuid(duck_value) };
91        assert_eq!(raw.lower, value.0 as u64);
92        assert_eq!(raw.upper, (value.0 >> 64) as u64);
93        // SAFETY: `duck_value` was created by `to_duck` above; destroyed exactly once.
94        unsafe { duckdb_destroy_value(&mut duck_value) };
95    }
96
97    /// `from_duck` reads the packed physical column representation directly (bypassing
98    /// the value API), which — unlike `to_duck`/`duckdb_get_uuid` — *is* sign-flipped.
99    /// Simulate that physical representation directly.
100    fn from_duck_undoes_storage_flip(value: DuckUuid) {
101        let physical = duckdb_uhugeint {
102            lower: value.0 as u64,
103            upper: ((value.0 >> 64) as u64) ^ UUID_SIGN_FLIP,
104        };
105        let converted = DuckUuid::from_duck(physical).unwrap();
106        assert_eq!(value, converted);
107    }
108
109    #[test]
110    fn nil_uuid() {
111        to_duck_is_passthrough(DuckUuid(0));
112        from_duck_undoes_storage_flip(DuckUuid(0));
113    }
114
115    #[test]
116    fn max_uuid() {
117        to_duck_is_passthrough(DuckUuid(u128::MAX));
118        from_duck_undoes_storage_flip(DuckUuid(u128::MAX));
119    }
120
121    #[test]
122    fn arbitrary_uuid() {
123        let v = DuckUuid(0x1234_5678_9abc_def0_0fed_cba9_8765_4321);
124        to_duck_is_passthrough(v);
125        from_duck_undoes_storage_flip(v);
126    }
