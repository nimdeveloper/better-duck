10/// A DuckDB `BLOB` value — a raw byte sequence of arbitrary length.
11///
12/// Use `Blob` anywhere you need to read or write a DuckDB `BLOB` column. The inner
13/// `Vec<u8>` is always an owned copy of the bytes stored in DuckDB.
14///
15/// `DuckValue::Blob` still holds `Vec<u8>` directly; this new type is provided so that
16/// a plain `Vec<u8>` is **not** forced to map to `BLOB`, freeing `Vec<T>` for use in
17/// a generic `LIST` `From` impl
18#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
19pub struct Blob(pub Vec<u8>);
20
21impl Blob {
22    /// Creates a new `Blob` from a byte vector.
23    #[inline]
24    pub fn new(bytes: Vec<u8>) -> Self {
25        Blob(bytes)
26    }
27
28    /// Returns a slice of the blob's bytes.
29    #[inline]
30    pub fn as_bytes(&self) -> &[u8] {
31        &self.0
32    }
33}
34
35impl From<Vec<u8>> for Blob {
36    fn from(v: Vec<u8>) -> Self {
37        Blob(v)
38    }
39}

41impl From<Blob> for Vec<u8> {
42    fn from(b: Blob) -> Self {
43        b.0
44    }
45}

47impl DuckDialect for Blob {
48    fn from_duck(value: duckdb_value) -> Result<Self, DuckDBConversionError> {
49        // SAFETY: `value` is a valid duckdb_value of type BLOB.
50        // `duckdb_get_blob` returns a pointer + size; we copy the bytes immediately
51        // and do not retain the raw pointer after this block.
52        let blob = unsafe { duckdb_get_blob(value) };
53        if blob.data.is_null() {
54            return Err(DuckDBConversionError::NullValue);
55        }
56        // SAFETY: `blob.data` is a valid pointer to `blob.size` bytes for the duration
57        // of this call; we copy immediately.
58        let slice =
59            unsafe { std::slice::from_raw_parts(blob.data as *const u8, blob.size as usize) };
60        Ok(Blob(slice.to_vec()))
61    }
62
63    fn to_duck(&self) -> Result<duckdb_value, DuckDBConversionError> {
64        // SAFETY: `self.0.as_ptr()` is valid for `self.0.len()` bytes.
65        // `duckdb_create_blob` copies the bytes internally.
66        Ok(unsafe { duckdb_create_blob(self.0.as_ptr(), self.0.len() as crate::ffi::idx_t) })
67    }
68}

70impl AppendAble for Blob {
71    fn appender_append(
72        &mut self,
73        appender: crate::ffi::duckdb_appender,
74    ) -> crate::error::Result<()> {
75        // SAFETY: `self.0.as_ptr()` is valid for `self.0.len()` bytes; append copies the data.
76        unsafe {
77            crate::ffi::duckdb_append_blob(
78                appender,
79                self.0.as_ptr() as *const std::ffi::c_void,
80                self.0.len() as u64,
81            )
82        };
83        Ok(())
84    }
85
86    fn stmt_append(
87        &mut self,
88        idx: u64,
89        stmt: crate::ffi::duckdb_prepared_statement,
90    ) -> crate::error::Result<()> {
91        // SAFETY: `self.0.as_ptr()` is valid for `self.0.len()` bytes; bind copies the data.
92        unsafe {
93            crate::ffi::duckdb_bind_blob(
94                stmt,
95                idx,
96                self.0.as_ptr() as *const std::ffi::c_void,
97                self.0.len() as u64,
98            )
99        };
100        Ok(())
101    }
102}

104impl DuckLogicalType for Blob {
105    fn duck_logical_type() -> Result<duckdb_logical_type, DuckDBConversionError> {
106        // SAFETY: DUCKDB_TYPE_DUCKDB_TYPE_BLOB is always a valid duckdb_type constant.
107        Ok(unsafe { duckdb_create_logical_type(DUCKDB_TYPE_DUCKDB_TYPE_BLOB) })
108    }
109}

131    fn assert_ffi_round_trip(expected: Blob) {
132        let mut raw = expected.to_duck().unwrap();
133        let actual = Blob::from_duck(raw);
134        // SAFETY: `raw` was created by `to_duck` above and is destroyed exactly once here.
135        unsafe { duckdb_destroy_value(&mut raw) };
136        assert_eq!(actual.unwrap(), expected);
137    }

139    #[test]
140    fn constructor_access_and_standard_conversions_preserve_bytes() {
141        let bytes = vec![0, 1, 0, 255];
142        let blob = Blob::new(bytes.clone());
143        assert_eq!(blob.as_bytes(), bytes.as_slice());
144
145        let from_vec = Blob::from(bytes.clone());
146        let back_to_vec: Vec<u8> = from_vec.into();
147        assert_eq!(back_to_vec, bytes);
148    }

150    #[test]
151    fn ordering_and_hash_follow_byte_contents() {
152        let first = Blob::new(vec![0]);
153        let second = Blob::new(vec![0, 1]);
154        let third = Blob::new(vec![1]);
155
156        let ordered = BTreeSet::from([third.clone(), first.clone(), second.clone()]);
157        assert_eq!(ordered.into_iter().collect::<Vec<_>>(), vec![first.clone(), second, third]);
158
159        let mut hashed = HashSet::new();
160        assert!(hashed.insert(first.clone()));
161        assert!(!hashed.insert(first.clone()));
162        assert!(hashed.contains(&first));
163    }

191    #[test]
192    fn prepared_statement_and_appender_preserve_binary_blobs() {
193        let mut conn = Connection::open_in_memory().unwrap();
194        conn.execute_batch("CREATE TABLE blobs (value BLOB)").unwrap();
195
196        let mut prepared_blob = Blob::new(vec![0, 1, 0, 255]);
197        conn.execute_with("INSERT INTO blobs VALUES ($1)", &mut [&mut prepared_blob]).unwrap();
198
199        {
200            let mut appender = conn.appender("blobs", "main").unwrap();
201            let mut appender_blob = Blob::new(vec![255, 0, 2, 0]);
202            appender.append(&mut appender_blob).unwrap();
203            appender.save().unwrap();
204        }
205
206        let values = conn
207            .execute("SELECT value FROM blobs ORDER BY value")
208            .unwrap()
209            .map(|row| row.unwrap().get("value").unwrap().clone())
210            .collect::<Vec<_>>();
211
212        assert_eq!(
213            values,
214            vec![
215                DuckValue::Blob(Blob::new(vec![0, 1, 0, 255])),
