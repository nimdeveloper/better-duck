1//! MAP read/write helpers + [`AppendAble`] impl for `HashMap<DuckValue, DuckValue>`.
2//!
3//! DuckDB `MAP` is physically stored as `LIST<STRUCT(key, value)>`.  The read path
4//! walks the flat STRUCT child vectors using the same validity + recursion pattern as
5//! LIST/ARRAY.  The write path builds a `duckdb_value` via `duckdb_create_map_value`.
6//!
7//! Unlike the previous string-keyed representation, MAP keys are now real `DuckValue`
8//! values, preserving the full DuckDB key type.
9// FFI pointer arguments are used safely inside `unsafe` blocks.
10#![allow(clippy::not_unsafe_ptr_arg_deref)]
11

29// Read path

31/// Read a DuckDB `MAP` vector column at `row_idx`.
32///
33/// MAP is physically `LIST<STRUCT(key, value)>`.  Keys are returned as real
34/// [`DuckValue`] instances — no stringification.
35///
36/// # Safety
37/// `val` must be a valid `duckdb_vector` of MAP type; `row_idx` must be within
38/// `[0, chunk_size)`.
39pub(crate) fn read_map(
40    val: duckdb_vector,
41    row_idx: u64,
42) -> Result<DuckValue, DuckDBConversionError> {
43    // SAFETY: MAP data layout is identical to LIST: each row slot holds a
44    // `duckdb_list_entry { offset, length }`.
45    let data_ptr = unsafe { duckdb_vector_get_data(val) as *const duckdb_list_entry };
46    // SAFETY: `row_idx` is within [0, chunk_size).
47    let entry: duckdb_list_entry = unsafe { *data_ptr.add(row_idx as usize) };
48
49    // `entries_vec` is the flat STRUCT(key, value) child vector.
50    // SAFETY: `val` is a valid MAP/LIST vector.
51    let entries_vec = unsafe { duckdb_list_vector_get_child(val) };
52    // SAFETY: `entries_vec` is a valid STRUCT vector; child 0 = keys.
53    let key_vec = unsafe { duckdb_struct_vector_get_child(entries_vec, 0) };
54    // SAFETY: `entries_vec` is a valid STRUCT vector; child 1 = values.
55    let val_vec = unsafe { duckdb_struct_vector_get_child(entries_vec, 1) };
56
57    // SAFETY: `key_vec` is a valid vector.
58    let mut key_lt = unsafe { duckdb_vector_get_column_type(key_vec) };
59    // SAFETY: `key_lt` was returned by `duckdb_vector_get_column_type`.
60    let key_tid = unsafe { duckdb_get_type_id(key_lt) };
61    // SAFETY: `key_lt` was allocated by `duckdb_vector_get_column_type` above.
62    unsafe { duckdb_destroy_logical_type(&mut key_lt) };
63
64    // SAFETY: `val_vec` is a valid vector.
65    let mut vlt = unsafe { duckdb_vector_get_column_type(val_vec) };
66    // SAFETY: `vlt` was returned by `duckdb_vector_get_column_type`.
67    let val_tid = unsafe { duckdb_get_type_id(vlt) };
68    // SAFETY: `vlt` was allocated by `duckdb_vector_get_column_type` above.
69    unsafe { duckdb_destroy_logical_type(&mut vlt) };
70
71    let mut pairs: HashMap<DuckValue, DuckValue> = HashMap::with_capacity(entry.length as usize);
72    let mut read_err: Option<DuckDBConversionError> = None;
73
74    for j in entry.offset..entry.offset + entry.length {
75        let k = match DuckValue::from_duckdb_vec(key_vec, key_tid, j) {
76            Ok(v) => v,
77            Err(e) => {
78                read_err = Some(e);
79                break;
80            },
81        };
82        let v = match DuckValue::from_duckdb_vec(val_vec, val_tid, j) {
83            Ok(v) => v,
84            Err(e) => {
85                read_err = Some(e);
86                break;
87            },
88        };
89        pairs.insert(k, v);
90    }
91
92    match read_err {
93        Some(e) => Err(e),
94        None => Ok(DuckValue::Map(pairs)),
95    }
96}

100/// Build a `duckdb_value` of type `MAP` from a `HashMap<DuckValue, DuckValue>`.
101///
102/// Returns an error for empty maps (value type cannot be inferred).
103/// The caller is responsible for destroying the returned value.
104pub(crate) fn map_to_duck(
105    m: &HashMap<DuckValue, DuckValue>
106) -> Result<duckdb_value, DuckDBConversionError> {
107    let n = m.len();
108    if n == 0 {
109        return Err(DuckDBConversionError::ConversionError(
110            "cannot convert empty Map to duckdb_value: value type unknown".into(),
111        ));
112    }
113    let pairs: Vec<(&DuckValue, &DuckValue)> = m.iter().collect();
114    let mut key_lt = DuckValue::logical_type_of(pairs[0].0)?; // mut needed for duckdb_destroy_logical_type
115    let mut val_lt = match DuckValue::logical_type_of(pairs[0].1) {
116        Ok(lt) => lt,
117        Err(e) => {
118            // SAFETY: `key_lt` was allocated above.
119            unsafe { duckdb_destroy_logical_type(&mut key_lt) };
120            return Err(e);
121        },
122    };
123    // SAFETY: both types are valid; `duckdb_create_map_type` copies them.
124    let mut map_lt = unsafe { duckdb_create_map_type(key_lt, val_lt) };
125    // SAFETY: `key_lt` was allocated by `logical_type_of` above; destroy once.
126    unsafe { duckdb_destroy_logical_type(&mut key_lt) };
127    // SAFETY: `val_lt` was allocated by `logical_type_of` above; destroy once.
128    unsafe { duckdb_destroy_logical_type(&mut val_lt) };
129
130    let mut key_dvs: Vec<duckdb_value> = Vec::with_capacity(n);
131    let mut val_dvs: Vec<duckdb_value> = Vec::with_capacity(n);
132    let mut err: Option<DuckDBConversionError> = None;
133    for (k, v) in &pairs {
134        match k.to_duck() {
135            Ok(kv) => key_dvs.push(kv),
136            Err(e) => {
137                err = Some(e);
138                break;
139            },
140        }
141        match v.to_duck() {
142            Ok(vv) => val_dvs.push(vv),
143            Err(e) => {
144                err = Some(e);
145                break;
146            },
147        }
148    }
149    if let Some(e) = err {
150        for mut kv in key_dvs {
151            // SAFETY: each `kv` was created by `to_duck()` above; destroy once.
152            unsafe { duckdb_destroy_value(&mut kv) };
153        }
154        for mut vv in val_dvs {
155            // SAFETY: each `vv` was created by `to_duck()` above; destroy once.
156            unsafe { duckdb_destroy_value(&mut vv) };
157        }
158        // SAFETY: `map_lt` was allocated above; destroy once.
159        unsafe { duckdb_destroy_logical_type(&mut map_lt) };
160        return Err(e);
161    }
162    // SAFETY: `map_lt` valid; key/val arrays have `n` elements each.
163    let result = unsafe {
164        duckdb_create_map_value(map_lt, key_dvs.as_mut_ptr(), val_dvs.as_mut_ptr(), n as idx_t)
165    };
166    // SAFETY: `map_lt` was allocated above; destroy once.
167    unsafe { duckdb_destroy_logical_type(&mut map_lt) };
168    for mut kv in key_dvs {
169        // SAFETY: each `kv` was created by `to_duck()` above; destroy once.
170        unsafe { duckdb_destroy_value(&mut kv) };
171    }
172    for mut vv in val_dvs {
173        // SAFETY: each `vv` was created by `to_duck()` above; destroy once.
174        unsafe { duckdb_destroy_value(&mut vv) };
175    }
176    Ok(result)
177}

179// Logical-type path

181/// Return a `duckdb_logical_type` for a MAP with key/value types inferred from the
182/// first entry.
183pub(crate) fn map_logical_type(
184    m: &HashMap<DuckValue, DuckValue>
185) -> Result<duckdb_logical_type, DuckDBConversionError> {
186    if m.is_empty() {
187        return Err(DuckDBConversionError::ConversionError(
188            "cannot determine value type of empty Map".into(),
189        ));
190    }
191    let (first_k, first_v) = m.iter().next().unwrap();
192    let mut key_lt = DuckValue::logical_type_of(first_k)?;
193    let mut val_lt = match DuckValue::logical_type_of(first_v) {
194        Ok(lt) => lt,
195        Err(e) => {
196            // SAFETY: `key_lt` was allocated by `logical_type_of` above; destroy once.
197            unsafe { duckdb_destroy_logical_type(&mut key_lt) };
198            return Err(e);
199        },
200    };
201    // SAFETY: both types are valid; `duckdb_create_map_type` copies them.
202    let lt = unsafe { duckdb_create_map_type(key_lt, val_lt) };
203    // SAFETY: `key_lt` was allocated by `logical_type_of` above; destroy once.
204    unsafe { duckdb_destroy_logical_type(&mut key_lt) };
205    // SAFETY: `val_lt` was allocated by `logical_type_of` above; destroy once.
206    unsafe { duckdb_destroy_logical_type(&mut val_lt) };
207    Ok(lt)
208}

210// AppendAble impl
211//
212// Unlike `map_to_duck` above (which infers key/value types from the first entry and
213// therefore rejects empty maps), this uses `K`/`V::duck_logical_type()` — a property
214// of the Rust types, not of any particular value — so it works for empty maps too.
215//
216// This generic impl covers `HashMap<DuckValue, DuckValue>` too (`DuckValue` does not
217// implement `DuckLogicalType`, but a caller with an actual `HashMap<DuckValue,
218// DuckValue>` can bind `DuckValue::Map(m)` directly, which goes through the
219// value-based `map_to_duck` path above instead).
220//
221// A plain `HashMap<String, DuckValue>` therefore means MAP here; wrap it in
222// `DuckStruct` (see `duck_struct.rs`) to mean STRUCT instead — mirroring how `Blob`
223// disambiguates `Vec<u8>` from a generic `LIST`.
224
225/// Builds a `duckdb_value` of type `MAP` from `m`, using `K`/`V`'s static logical
226/// types for the key/value types. Works even when `m` is empty.
227fn build_typed_map_value<K, V>(m: &HashMap<K, V>) -> Result<duckdb_value, DuckDBConversionError>
228where
229    K: Into<DuckValue> + Clone + DuckLogicalType,
230    V: Into<DuckValue> + Clone + DuckLogicalType,
231{
232    let mut key_lt = K::duck_logical_type()?;
233    let mut val_lt = match V::duck_logical_type() {
234        Ok(lt) => lt,
235        Err(e) => {
236            // SAFETY: `key_lt` was allocated by `K::duck_logical_type()` above.
237            unsafe { duckdb_destroy_logical_type(&mut key_lt) };
238            return Err(e);
239        },
240    };
241    // SAFETY: both types are valid; `duckdb_create_map_type` copies them.
242    let mut map_lt = unsafe { duckdb_create_map_type(key_lt, val_lt) };
243    // SAFETY: `key_lt` was allocated above; destroy once.
244    unsafe { duckdb_destroy_logical_type(&mut key_lt) };
245    // SAFETY: `val_lt` was allocated above; destroy once.
246    unsafe { duckdb_destroy_logical_type(&mut val_lt) };
247
248    let n = m.len();
249    let mut key_dvs: Vec<duckdb_value> = Vec::with_capacity(n);
250    let mut val_dvs: Vec<duckdb_value> = Vec::with_capacity(n);
251    let mut err: Option<DuckDBConversionError> = None;
252    for (k, v) in m {
253        match k.clone().into().to_duck() {
254            Ok(kv) => key_dvs.push(kv),
255            Err(e) => {
256                err = Some(e);
257                break;
258            },
259        }
260        match v.clone().into().to_duck() {
261            Ok(vv) => val_dvs.push(vv),
262            Err(e) => {
263                err = Some(e);
264                break;
265            },
266        }
267    }
268    if let Some(e) = err {
269        for mut kv in key_dvs {
270            // SAFETY: each `kv` was created by `to_duck()` above; destroy once.
271            unsafe { duckdb_destroy_value(&mut kv) };
272        }
273        for mut vv in val_dvs {
274            // SAFETY: each `vv` was created by `to_duck()` above; destroy once.
275            unsafe { duckdb_destroy_value(&mut vv) };
276        }
277        // SAFETY: `map_lt` was allocated above; destroy once.
278        unsafe { duckdb_destroy_logical_type(&mut map_lt) };
279        return Err(e);
280    }
281    // SAFETY: `map_lt` valid; key/val arrays have `n` elements each.
282    let result = unsafe {
283        duckdb_create_map_value(map_lt, key_dvs.as_mut_ptr(), val_dvs.as_mut_ptr(), n as idx_t)
284    };
285    // SAFETY: `map_lt` was allocated above; destroy once.
286    unsafe { duckdb_destroy_logical_type(&mut map_lt) };
287    for mut kv in key_dvs {
288        // SAFETY: each `kv` was created by `to_duck()` above; destroy once.
289        unsafe { duckdb_destroy_value(&mut kv) };
290    }
291    for mut vv in val_dvs {
292        // SAFETY: each `vv` was created by `to_duck()` above; destroy once.
293        unsafe { duckdb_destroy_value(&mut vv) };
294    }
295    Ok(result)
296}

298/// Bind/append a `HashMap<K, V>` as a DuckDB `MAP`, for any `K`/`V` with a fixed
299/// DuckDB type (see [`DuckLogicalType`]). Works for empty maps too.
300impl<K, V> AppendAble for HashMap<K, V>
301where
302    K: Into<DuckValue> + Clone + DuckLogicalType,
303    V: Into<DuckValue> + Clone + DuckLogicalType,
304{
305    fn stmt_append(
306        &mut self,
307        idx: u64,
308        stmt: crate::ffi::duckdb_prepared_statement,
309    ) -> Result<()> {
310        let mut dv = build_typed_map_value(self).map_err(Error::ConversionError)?;
311        // SAFETY: `stmt`/`idx` are valid; `dv` was created by `build_typed_map_value`.
312        unsafe { duckdb_bind_value(stmt, idx, dv) };
313        // SAFETY: `dv` was created above; destroy exactly once.
314        unsafe { duckdb_destroy_value(&mut dv) };
315        Ok(())
316    }
317
318    fn appender_append(
319        &mut self,
320        appender: crate::ffi::duckdb_appender,
321    ) -> Result<()> {
322        let mut dv = build_typed_map_value(self).map_err(Error::ConversionError)?;
323        // SAFETY: `appender` is valid; `dv` was created by `build_typed_map_value`.
324        unsafe { duckdb_append_value(appender, dv) };
325        // SAFETY: `dv` was created above; destroy exactly once.
326        unsafe { duckdb_destroy_value(&mut dv) };
327        Ok(())
328    }
329}

355/// Order-independent hash for map/struct entries: XOR of per-entry hashes.
356///
357/// We hash each `(k, v)` pair with a fresh `DefaultHasher`, then XOR-fold the
358/// results so that insertion order does not affect the combined hash.
359pub(super) fn map_entries_hash<H: Hasher, K: Hash, V: Hash>(
360    entries: impl Iterator<Item = (K, V)>,
361    len: usize,
362    state: &mut H,
363) {
364    len.hash(state);
365    let xor_fold: u64 = entries
366        .map(|(k, v)| {
367            let mut h = DefaultHasher::new();
368            k.hash(&mut h);
369            v.hash(&mut h);
370            h.finish()
371        })
372        .fold(0u64, |acc, x| acc ^ x);
373    xor_fold.hash(state);
374}

376/// Order-independent hash for map/struct entries.
377pub(super) fn map_entries_hash_ref<'a, H: Hasher>(
378    entries: impl Iterator<Item = (&'a DuckValueRef<'a>, &'a DuckValueRef<'a>)>,
379    len: usize,
380    state: &mut H,
381) where
382    DuckValueRef<'a>: Hash,
383{
384    len.hash(state);
385    let xor_fold: u64 = entries
386        .map(|(k, v)| {
387            let mut h = DefaultHasher::new();
388            k.hash(&mut h);
389            v.hash(&mut h);
390            h.finish()
391        })
392        .fold(0u64, |acc, x| acc ^ x);
393    xor_fold.hash(state);
394}

401    #[test]
402    fn empty_hashmap_appends_and_reads_back() {
403        let mut conn = Connection::open_in_memory().unwrap();
404        conn.execute_batch("CREATE TABLE t (v MAP(INTEGER, VARCHAR))").unwrap();
405        let mut map: HashMap<i32, String> = HashMap::new();
406        conn.execute_with("INSERT INTO t VALUES ($1)", &mut [&mut map]).unwrap();
407        let mut result = conn.execute("SELECT v FROM t").unwrap();
408        let row = result.next().unwrap().unwrap();
409        match row.get("v").unwrap() {
410            DuckValue::Map(m) => assert!(m.is_empty()),
411            other => panic!("expected Map, got {other:?}"),
412        }

417        let mut conn = Connection::open_in_memory().unwrap();
