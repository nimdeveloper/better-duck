29/// Read a DuckDB `LIST` or `ARRAY` vector column at `row_idx`.
30///
31/// Dispatched from [`DuckValue::from_duckdb_vec`] for the `LIST` and `ARRAY` type ids.
32///
33/// # Safety
34/// `val` must be a valid `duckdb_vector` of LIST or ARRAY type; `row_idx` must be
35/// within `[0, chunk_size)`.
36#[inline(always)]
37pub(crate) fn read_list_or_array(
38    val: duckdb_vector,
39    t: duckdb_type,
40    row_idx: u64,
41) -> Result<DuckValue, DuckDBConversionError> {
42    // Compute the (offset, length) of child elements for this row.
43    //
44    // LIST: the parent vector stores a `duckdb_list_entry` per row with offset+length.
45    // ARRAY (fixed-size): no `duckdb_list_entry` exists — `duckdb_vector_get_data` returns
46    // null for ARRAY columns. Instead we read the fixed size from the parent logical type
47    // and compute offset = row_idx * size.
48    let (offset, length) = if t == DUCKDB_TYPE_DUCKDB_TYPE_LIST {
49        // SAFETY: `val` is a valid LIST vector; `duckdb_vector_get_data` returns a pointer
50        // to a packed `duckdb_list_entry[]` with at least `chunk_size` entries.
51        let data_ptr = unsafe { duckdb_vector_get_data(val) as *mut duckdb_list_entry };
52        // SAFETY: `data_ptr` is a valid non-null pointer; `row_idx` is within [0, chunk_size).
53        let entry = unsafe { *data_ptr.add(row_idx as usize) };
54        (entry.offset, entry.length)
55    } else {
56        // SAFETY: `val` is a valid ARRAY vector; `duckdb_vector_get_column_type` always
57        // succeeds for a non-null vector.
58        let mut parent_lt = unsafe { duckdb_vector_get_column_type(val) };
59        // SAFETY: `parent_lt` is a valid logical type for an ARRAY column.
60        let array_size = unsafe { duckdb_array_type_array_size(parent_lt) };
61        // SAFETY: `parent_lt` was obtained above and must be freed exactly once.
62        unsafe { duckdb_destroy_logical_type(&mut parent_lt) };
63        (row_idx * array_size as u64, array_size as u64)
64    };
65
66    // SAFETY: `val` is a valid duckdb_vector of list/array type.
67    let list_child = unsafe {
68        if t == DUCKDB_TYPE_DUCKDB_TYPE_LIST {
69            duckdb_list_vector_get_child(val)
70        } else {
71            duckdb_array_vector_get_child(val)
72        }
73    };
74
75    // SAFETY: `list_child` is a valid duckdb_vector for the child column.
76    let child_validity = unsafe { duckdb_vector_get_validity(list_child) };
77
78    // SAFETY: `list_child` is a valid duckdb_vector; `duckdb_vector_get_column_type` always
79    // succeeds for a valid vector and returns an owned logical type.
80    let mut raw_child_type: duckdb_logical_type =
81        unsafe { duckdb_vector_get_column_type(list_child) };
82    // SAFETY: `raw_child_type` is a valid logical type returned by `duckdb_vector_get_column_type`.
83    let child_type = unsafe { duckdb_get_type_id(raw_child_type) };
84    // SAFETY: `raw_child_type` was obtained above and must be freed exactly once.
85    unsafe { duckdb_destroy_logical_type(&mut raw_child_type) };
86
87    let mut slice_data: Option<Box<[std::mem::MaybeUninit<DuckValue>]>> = None;
88    let mut vec_data: Option<Vec<DuckValue>> = None;
89    let iter_ptr: *mut DuckValue;
90
91    if t == DUCKDB_TYPE_DUCKDB_TYPE_ARRAY {
92        let mut tmp = Box::<[DuckValue]>::new_uninit_slice(length as usize);
93        iter_ptr = tmp.as_mut_ptr() as *mut DuckValue;
94        slice_data = Some(tmp);
95    } else if t == DUCKDB_TYPE_DUCKDB_TYPE_LIST {
96        let mut tmp = Vec::with_capacity(length as usize);
97        iter_ptr = tmp.as_mut_ptr();
98        vec_data = Some(tmp);
99    } else {
100        return Err(DuckDBConversionError::ConversionError(
101            "invalid type for list/array".to_owned(),
102        ));
103    }
104
105    // SAFETY: `offset` and `length` describe the range of child elements for this row.
106    // `iter_ptr` points to an allocation of exactly `length` elements.
107    // `i` is the relative (0-based) write index; `each` is the absolute child-vector index.
108    unsafe {
109        for (i, each) in (offset..(offset + length)).enumerate() {
110            let mut elem = DuckValue::Null;
111            if duckdb_validity_row_is_valid(child_validity, each) {
112                elem = DuckValue::from_duckdb_vec(list_child, child_type, each)?;
113            }
114            ptr::write(iter_ptr.add(i), elem);
115        }
116    };
117
118    if t == DUCKDB_TYPE_DUCKDB_TYPE_ARRAY {
119        // SAFETY: every element in `slice_data` was written in the loop above.
120        Ok(DuckValue::Array(unsafe { slice_data.unwrap().assume_init() }))
121    } else {
122        let mut vec_data = vec_data.unwrap();
123        // SAFETY: all `length` elements were written into `vec_data`'s allocation
124        // via `iter_ptr` in the loop above.
125        unsafe { vec_data.set_len(length as usize) };
126        Ok(DuckValue::List(vec_data))
127    }
128}

132/// Build a `duckdb_value` of type `LIST` from a Rust slice.
133///
134/// Returns an error for empty slices (element type cannot be inferred).
135/// The caller is responsible for destroying the returned value with `duckdb_destroy_value`.
136pub(crate) fn list_to_duck(items: &[DuckValue]) -> Result<duckdb_value, DuckDBConversionError> {
137    if items.is_empty() {
138        return Err(DuckDBConversionError::ConversionError(
139            "cannot convert empty List to duckdb_value: element type unknown".into(),
140        ));
141    }
142    let mut child_lt = DuckValue::logical_type_of(&items[0])?;
143    let mut child_dvs: Vec<duckdb_value> = Vec::with_capacity(items.len());
144    let mut err: Option<DuckDBConversionError> = None;
145    for item in items {
146        match item.to_duck() {
147            Ok(v) => child_dvs.push(v),
148            Err(e) => {
149                err = Some(e);
150                break;
151            },
152        }
153    }
154    if let Some(e) = err {
155        for mut v in child_dvs {
156            // SAFETY: each `v` was created by `to_duck()` above.
157            unsafe { duckdb_destroy_value(&mut v) };
158        }
159        // SAFETY: `child_lt` was allocated by `logical_type_of` above.
160        unsafe { duckdb_destroy_logical_type(&mut child_lt) };
161        return Err(e);
162    }
163    // SAFETY: `child_lt` is valid; `child_dvs` has `len()` elements.
164    let result = unsafe {
165        duckdb_create_list_value(child_lt, child_dvs.as_mut_ptr(), child_dvs.len() as idx_t)
166    };
167    // SAFETY: `child_lt` was allocated by `logical_type_of`; destroy once.
168    unsafe { duckdb_destroy_logical_type(&mut child_lt) };
169    for mut v in child_dvs {
170        // SAFETY: each `v` was created by `to_duck()` above.
171        unsafe { duckdb_destroy_value(&mut v) };
172    }
173    Ok(result)
174}

177///
178/// Returns an error for empty slices.
179/// The caller is responsible for destroying the returned value.
180pub(crate) fn array_to_duck(items: &[DuckValue]) -> Result<duckdb_value, DuckDBConversionError> {
181    if items.is_empty() {
182        return Err(DuckDBConversionError::ConversionError(
183            "cannot convert empty Array to duckdb_value: element type unknown".into(),
184        ));
185    }
186    let mut child_lt = DuckValue::logical_type_of(&items[0])?;
187    let mut child_dvs: Vec<duckdb_value> = Vec::with_capacity(items.len());
188    let mut err: Option<DuckDBConversionError> = None;
189    for item in items {
190        match item.to_duck() {
191            Ok(v) => child_dvs.push(v),
192            Err(e) => {
193                err = Some(e);
194                break;
195            },
196        }
197    }
198    if let Some(e) = err {
199        for mut v in child_dvs {
200            // SAFETY: each `v` was created by `to_duck()` above.
201            unsafe { duckdb_destroy_value(&mut v) };
202        }
203        // SAFETY: `child_lt` was allocated by `logical_type_of` above.
204        unsafe { duckdb_destroy_logical_type(&mut child_lt) };
205        return Err(e);

215    unsafe { duckdb_destroy_logical_type(&mut child_lt) };
216    for mut v in child_dvs {
217        // SAFETY: each `v` was created by `to_duck()` above.
218        unsafe { duckdb_destroy_value(&mut v) };
219    }
220    Ok(result)
221}

225/// Return a `duckdb_logical_type` for a LIST of the element type of `items[0]`.
226pub(crate) fn list_logical_type(
227    items: &[DuckValue]
228) -> Result<duckdb_logical_type, DuckDBConversionError> {
229    if items.is_empty() {
230        return Err(DuckDBConversionError::ConversionError(
231            "cannot determine element type of empty List".into(),
232        ));
233    }
234    let mut child_lt = DuckValue::logical_type_of(&items[0])?;
235    // SAFETY: `child_lt` is a valid logical type; `duckdb_create_list_type` copies it.
236    let lt = unsafe { duckdb_create_list_type(child_lt) };
237    // SAFETY: `child_lt` was allocated above and must be freed exactly once.
238    unsafe { duckdb_destroy_logical_type(&mut child_lt) };
239    Ok(lt)
240}

259// Generic AppendAble impls
260//
261// Unlike `list_to_duck`/`array_to_duck` above (which infer the element type from
262// `items[0]` and therefore reject empty collections), these use `T::duck_logical_type()`
263// — a property of the Rust type, not of any particular value — so they work for
264// empty `Vec<T>`/`Box<[T]>` too.
265
266/// Builds a `duckdb_value` of type `LIST` from `items`, using `T`'s static logical
267/// type for the element type. Works even when `items` is empty.
268fn build_typed_list_value<T: Into<DuckValue> + Clone + DuckLogicalType>(
269    items: &[T]
270) -> Result<duckdb_value, DuckDBConversionError> {
271    let mut child_lt = T::duck_logical_type()?;
272    let mut child_dvs: Vec<duckdb_value> = Vec::with_capacity(items.len());
273    let mut err: Option<DuckDBConversionError> = None;
274    for item in items {
275        match item.clone().into().to_duck() {
276            Ok(v) => child_dvs.push(v),
277            Err(e) => {
278                err = Some(e);
279                break;
280            },
281        }
282    }
283    if let Some(e) = err {
284        for mut v in child_dvs {
285            // SAFETY: each `v` was created by `to_duck()` above.
286            unsafe { duckdb_destroy_value(&mut v) };
287        }
288        // SAFETY: `child_lt` was allocated by `T::duck_logical_type()` above.
289        unsafe { duckdb_destroy_logical_type(&mut child_lt) };
290        return Err(e);
291    }
292    // SAFETY: `child_lt` is valid; `child_dvs` has `len()` elements.
293    let result = unsafe {
294        duckdb_create_list_value(child_lt, child_dvs.as_mut_ptr(), child_dvs.len() as idx_t)
295    };
296    // SAFETY: `child_lt` was allocated above; destroy once.
297    unsafe { duckdb_destroy_logical_type(&mut child_lt) };
298    for mut v in child_dvs {
299        // SAFETY: each `v` was created by `to_duck()` above; destroy once.
300        unsafe { duckdb_destroy_value(&mut v) };
301    }
302    Ok(result)
303}

305/// Builds a `duckdb_value` of type `ARRAY` from `items`, using `T`'s static logical
306/// type for the element type.
307///
308/// Unlike LIST, DuckDB's C API hard-asserts `array_size > 0` inside
309/// `duckdb_create_array_value` (aborting the process, not returning an error), so an
310/// empty `items` is rejected up front with a proper `Err` instead.
311fn build_typed_array_value<T: Into<DuckValue> + Clone + DuckLogicalType>(
312    items: &[T]
313) -> Result<duckdb_value, DuckDBConversionError> {
314    if items.is_empty() {
315        return Err(DuckDBConversionError::ConversionError(
316            "cannot convert empty slice to a duckdb ARRAY value: DuckDB requires ARRAY size >= 1"
317                .into(),
318        ));
319    }
320    let mut child_lt = T::duck_logical_type()?;
321    let mut child_dvs: Vec<duckdb_value> = Vec::with_capacity(items.len());
322    let mut err: Option<DuckDBConversionError> = None;
323    for item in items {
324        match item.clone().into().to_duck() {
325            Ok(v) => child_dvs.push(v),
326            Err(e) => {
327                err = Some(e);
328                break;
329            },
330        }
331    }
332    if let Some(e) = err {
333        for mut v in child_dvs {
334            // SAFETY: each `v` was created by `to_duck()` above.
335            unsafe { duckdb_destroy_value(&mut v) };
336        }
337        // SAFETY: `child_lt` was allocated by `T::duck_logical_type()` above.
338        unsafe { duckdb_destroy_logical_type(&mut child_lt) };
339        return Err(e);
340    }
341    // SAFETY: `child_lt` is valid; `duckdb_create_array_value` takes the *element* type
342    // (like `duckdb_create_list_value`), not a pre-built ARRAY type — it derives the
343    // array size from `value_count` itself (0 is valid). Passing a wrapped ARRAY type
344    // here would make DuckDB expect each child value to itself be an ARRAY, corrupting
345    // memory.
346    let result = unsafe {
347        duckdb_create_array_value(child_lt, child_dvs.as_mut_ptr(), child_dvs.len() as idx_t)
348    };
349    // SAFETY: `child_lt` was allocated above; destroy once.
350    unsafe { duckdb_destroy_logical_type(&mut child_lt) };
351    for mut v in child_dvs {
352        // SAFETY: each `v` was created by `to_duck()` above; destroy once.
353        unsafe { duckdb_destroy_value(&mut v) };
354    }
355    Ok(result)
356}

358/// Bind/append a `Vec<T>` as a DuckDB `LIST`, for any `T` with a fixed DuckDB type
359/// (see [`DuckLogicalType`]). Works for empty vectors too.
360impl<T: Into<DuckValue> + Clone + DuckLogicalType> AppendAble for Vec<T> {
361    fn stmt_append(
362        &mut self,
363        idx: u64,
364        stmt: crate::ffi::duckdb_prepared_statement,
365    ) -> Result<()> {
366        let mut dv = build_typed_list_value(self).map_err(Error::ConversionError)?;
367        // SAFETY: `stmt`/`idx` are valid; `dv` was created by `build_typed_list_value`.
368        unsafe { duckdb_bind_value(stmt, idx, dv) };
369        // SAFETY: `dv` was created above; destroy exactly once.
370        unsafe { duckdb_destroy_value(&mut dv) };
371        Ok(())
372    }
373
374    fn appender_append(
375        &mut self,
376        appender: crate::ffi::duckdb_appender,
377    ) -> Result<()> {
378        let mut dv = build_typed_list_value(self).map_err(Error::ConversionError)?;
379        // SAFETY: `appender` is valid; `dv` was created by `build_typed_list_value`.
380        unsafe { duckdb_append_value(appender, dv) };
381        // SAFETY: `dv` was created above; destroy exactly once.
382        unsafe { duckdb_destroy_value(&mut dv) };
383        Ok(())
384    }
385}

387/// Bind/append a `Box<[T]>` as a DuckDB `ARRAY`, for any `T` with a fixed DuckDB type
