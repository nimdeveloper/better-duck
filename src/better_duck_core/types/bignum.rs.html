1//! [`DuckDialect`] implementation for DuckDB's `BIGNUM` (arbitrary-precision integer) type.
2//!
3//! `BIGNUM` was previously named `VARINT` in some DuckDB releases; this crate targets
4//! the bundled DuckDB version, which uses `BIGNUM` (`duckdb_get_bignum` /
5//! `duckdb_create_bignum`).

17/// A DuckDB `BIGNUM` value: an arbitrary-precision signed integer.
18///
19/// `magnitude` holds the absolute value's bytes in little-endian order, matching
20/// `duckdb_bignum.data` exactly. `is_negative` is `true` for negative values.
21#[derive(Debug, Clone, PartialEq, Eq, Hash)]
22pub struct DuckBignum {
23    /// The absolute value, little-endian byte order.
24    pub magnitude: Vec<u8>,
25    /// `true` if this value is negative.
26    pub is_negative: bool,
27}
28
29impl DuckBignum {
30    /// Creates a `DuckBignum` from its little-endian magnitude bytes and sign.
31    #[inline]
32    pub fn new(
33        magnitude: Vec<u8>,
34        is_negative: bool,
35    ) -> DuckBignum {
36        DuckBignum { magnitude, is_negative }
37    }
38}

40impl DuckDialect for DuckBignum {
41    fn from_duck(value: duckdb_value) -> Result<Self, DuckDBConversionError> {
42        // SAFETY: `value` is a valid duckdb_value of type BIGNUM.
43        let raw = unsafe { duckdb_get_bignum(value) };
44        if raw.data.is_null() {
45            return Err(DuckDBConversionError::NullValue);
46        }
47        // SAFETY: `raw.data` is valid for `raw.size` bytes for the duration of this call;
48        // we copy immediately before freeing.
49        let magnitude = unsafe { std::slice::from_raw_parts(raw.data, raw.size as usize) }.to_vec();
50        // SAFETY: `raw.data` was allocated by DuckDB (`duckdb_get_bignum`) and must be
51        // freed with `duckdb_free` exactly once.
52        unsafe { duckdb_free(raw.data as *mut std::ffi::c_void) };
53        Ok(DuckBignum { magnitude, is_negative: raw.is_negative })
54    }
55
56    fn to_duck(&self) -> Result<duckdb_value, DuckDBConversionError> {
57        // `duckdb_create_bignum` does not tolerate a zero-length buffer (it overruns a
58        // stack buffer internally), so an empty magnitude — the value zero — is encoded
59        // as a single zero byte instead.
60        let zero = [0u8];
61        let (ptr, len) = if self.magnitude.is_empty() {
62            (zero.as_ptr(), 1)
63        } else {
64            (self.magnitude.as_ptr(), self.magnitude.len())
65        };
66        let bignum = duckdb_bignum {
67            // SAFETY-relevant: `duckdb_create_bignum` copies the bytes internally; the
68            // pointer only needs to stay valid for the duration of this call.
69            data: ptr as *mut u8,
70            size: len as crate::ffi::idx_t,
71            is_negative: self.is_negative,
72        };
73        // SAFETY: `bignum.data` is valid for `bignum.size` bytes for the duration of
74        // this call.
75        Ok(unsafe { duckdb_create_bignum(bignum) })
76    }
77}

94/// Decodes DuckDB's internal sortable wire format for `BIGNUM` physical column
95/// storage — used when reading directly from a data chunk vector (`from_duckdb_vec`
96/// in `value.rs`), which is a *different* representation from the `duckdb_value` API
97/// (`duckdb_get_bignum`/`duckdb_create_bignum`) used by [`DuckDialect::from_duck`]/
98/// [`DuckDialect::to_duck`] above.
99///
100/// Format (reverse-engineered by comparing `to_duck` input against the bytes observed
101/// after a real `INSERT` + chunk-vector read — DuckDB does not document this layout,
102/// so treat it as an internal, unversioned detail that could change in a future
103/// `libduckdb-sys` release): a 3-byte big-endian header `0x800000 | magnitude_len`,
104/// followed by `magnitude_len` bytes of magnitude in the same little-endian order as
105/// `duckdb_bignum.data`. For negative values, every byte (header and magnitude) is
106/// bitwise-complemented — this both flags the sign (the header's top bit becomes 0)
107/// and makes more-negative values sort before less-negative ones under a raw
108/// byte-wise comparison.
109pub(crate) fn decode_bignum_wire(raw: &[u8]) -> Option<DuckBignum> {
110    if raw.len() < 3 {
111        return None;
112    }
113    let is_negative = raw[0] & 0x80 == 0;
114    let header: u32 = if is_negative {
115        (u32::from(!raw[0]) << 16) | (u32::from(!raw[1]) << 8) | u32::from(!raw[2])
116    } else {
117        (u32::from(raw[0]) << 16) | (u32::from(raw[1]) << 8) | u32::from(raw[2])
118    };
119    let len = (header & 0x007F_FFFF) as usize;
120    let rest = &raw[3..];
121    if rest.len() < len {
122        return None;
123    }
124    let mut magnitude = rest[..len].to_vec();
125    if is_negative {
126        for b in &mut magnitude {
127            *b = !*b;
128        }
129    }
130    Some(DuckBignum { magnitude, is_negative })
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136    use crate::ffi::duckdb_destroy_value;
137
138    #[test]
139    fn decode_wire_positive() {
140        let decoded = decode_bignum_wire(&[128, 0, 2, 255, 1]).unwrap();
141        assert_eq!(decoded, DuckBignum::new(vec![255, 1], false));
142    }
143
144    #[test]
145    fn decode_wire_zero() {
146        let decoded = decode_bignum_wire(&[128, 0, 1, 0]).unwrap();
147        assert_eq!(decoded, DuckBignum::new(vec![0], false));
148    }
149
150    #[test]
151    fn decode_wire_negative() {
152        let decoded = decode_bignum_wire(&[127, 255, 254, 213]).unwrap();
153        assert_eq!(decoded, DuckBignum::new(vec![42], true));
154    }
155
156    #[test]
157    fn decode_wire_too_short_is_none() {
158        assert!(decode_bignum_wire(&[128, 0]).is_none());
159    }
160
161    #[test]
162    fn roundtrip_zero() {
163        let value = DuckBignum::new(vec![], false);
164        let mut duck_value = value.to_duck().unwrap();
165        let converted = DuckBignum::from_duck(duck_value).unwrap();
166        assert!(!converted.is_negative);
167        assert!(converted.magnitude.iter().all(|&b| b == 0));
168        // SAFETY: `duck_value` was created by `to_duck` above.
169        unsafe { duckdb_destroy_value(&mut duck_value) };
170    }
171
172    #[test]
173    fn roundtrip_positive() {
174        let value = DuckBignum::new(vec![0xFF, 0x01], false);
175        let mut duck_value = value.to_duck().unwrap();
176        let converted = DuckBignum::from_duck(duck_value).unwrap();
177        assert_eq!(value, converted);
178        // SAFETY: `duck_value` was created by `to_duck` above.
179        unsafe { duckdb_destroy_value(&mut duck_value) };
180    }
