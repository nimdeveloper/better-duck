13/// A DuckDB `BIT` (bitstring) value.
14///
15/// Stores the exact wire-format bytes DuckDB uses: the first byte holds the number
16/// of padding bits (0–7), and the remaining bytes hold the bit-packed data,
17/// most-significant bit first. This is the same layout as `duckdb_bit`, so a value
18/// read from DuckDB round-trips byte-for-byte without reinterpretation.
19#[derive(Debug, Clone, PartialEq, Eq, Hash)]
20pub struct DuckBit(pub Vec<u8>);
21
22impl DuckBit {
23    /// Wraps raw `BIT` wire-format bytes (padding-count byte + packed bit data).
24    #[inline]
25    pub fn new(bytes: Vec<u8>) -> DuckBit {
26        DuckBit(bytes)
27    }
28}
29
30impl DuckDialect for DuckBit {
31    fn from_duck(value: duckdb_value) -> Result<Self, DuckDBConversionError> {
32        // SAFETY: `value` is a valid duckdb_value of type BIT.
33        let raw = unsafe { duckdb_get_bit(value) };
34        if raw.data.is_null() {
35            return Err(DuckDBConversionError::NullValue);
36        }
37        // SAFETY: `raw.data` is valid for `raw.size` bytes for the duration of this call;
38        // we copy immediately before freeing.
39        let bytes = unsafe { std::slice::from_raw_parts(raw.data, raw.size as usize) }.to_vec();
40        // SAFETY: `raw.data` was allocated by DuckDB (`duckdb_get_bit`) and must be freed
41        // with `duckdb_free` exactly once.
42        unsafe { duckdb_free(raw.data as *mut std::ffi::c_void) };
43        Ok(DuckBit(bytes))
44    }
45
46    fn to_duck(&self) -> Result<duckdb_value, DuckDBConversionError> {
47        let bit = duckdb_bit {
48            // SAFETY-relevant: `duckdb_create_bit` copies the bytes internally; the pointer
49            // only needs to stay valid for the duration of this call.
50            data: self.0.as_ptr() as *mut u8,
51            size: self.0.len() as crate::ffi::idx_t,
52        };
53        // SAFETY: `bit.data` is valid for `bit.size` bytes for the duration of this call.
54        Ok(unsafe { duckdb_create_bit(bit) })
55    }
56}

60impl DuckLogicalType for DuckBit {
61    fn duck_logical_type() -> Result<duckdb_logical_type, DuckDBConversionError> {
62        // SAFETY: DUCKDB_TYPE_DUCKDB_TYPE_BIT is always a valid duckdb_type constant.
63        Ok(unsafe { duckdb_create_logical_type(DUCKDB_TYPE_DUCKDB_TYPE_BIT) })
64    }
65}

67impl From<DuckBit> for value::DuckValue {
68    fn from(b: DuckBit) -> Self {

73#[cfg(test)]
74mod tests {
75    use super::*;
76    use crate::ffi::duckdb_destroy_value;
77
78    #[test]
79    fn roundtrip_empty_bit() {
80        let value = DuckBit(vec![0u8]);
81        let mut duck_value = value.to_duck().unwrap();
82        let converted = DuckBit::from_duck(duck_value).unwrap();
83        assert_eq!(value, converted);
84        // SAFETY: `duck_value` was created by `to_duck` above.
85        unsafe { duckdb_destroy_value(&mut duck_value) };
86    }
87
88    #[test]
89    fn roundtrip_arbitrary_bits() {
90        let value = DuckBit(vec![3u8, 0b1110_0000]);
91        let mut duck_value = value.to_duck().unwrap();
92        let converted = DuckBit::from_duck(duck_value).unwrap();
93        assert_eq!(value, converted);
94        // SAFETY: `duck_value` was created by `to_duck` above.
95        unsafe { duckdb_destroy_value(&mut duck_value) };
96    }
