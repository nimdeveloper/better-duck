13/// A DuckDB appender for bulk-inserting rows into a table without going through
14/// the SQL parser.
15///
16/// Call [`append`](Appender::append) for each row and [`save`](Appender::save)
17/// to flush the data to the database. Rows are also flushed automatically on drop
18/// (errors during the implicit flush are logged to stderr).
19pub struct Appender {
20    _con: RawConnection,
21    inn: duckdb_appender,
22}
23
24impl Appender {
25    /// Creates a new `Appender` for the given table and schema.
26    ///
27    /// # Errors
28    ///
29    /// Returns an error if the table does not exist or the DuckDB appender cannot
30    /// be created.
31    pub fn new(
32        con: RawConnection,
33        table: &str,
34        schema: &str,
35    ) -> Result<Appender> {
36        let mut appender: duckdb_appender = ptr::null_mut();
37        let c_table = CString::new(table)?;
38        let c_schema = CString::new(schema)?;
39        // SAFETY: `con.con` is a valid open duckdb_connection. `c_schema` and `c_table`
40        // are valid null-terminated C strings. `appender` is a valid output pointer.
41        let res = unsafe {
42            duckdb_appender_create(
43                con.con,
44                c_schema.as_ptr() as *const c_char,
45                c_table.as_ptr() as *const c_char,
46                &mut appender,
47            )
48        };
49        result_from_duckdb_appender(res, &mut appender)
50            .map(|_| Appender { _con: con, inn: appender })
51    }
52
53    /// Appends a row to the table.
54    ///
55    /// Calls `duckdb_appender_begin_row`, then the value appender, then
56    /// `duckdb_appender_end_row`.
57    ///
58    /// # Errors
59    ///
60    /// Returns an error if the row cannot be appended.
61    #[must_use = "append result should be checked"]
62    #[allow(dead_code)]
63    pub fn append<T: AppendAble>(
64        &mut self,
65        row: &mut T,
66    ) -> Result<()> {
67        // SAFETY: `self.inn` is a valid duckdb_appender created in `new`.
68        let _ = unsafe { duckdb_appender_begin_row(self.inn) };
69        row.appender_append(self.inn)?;
70        // SAFETY: `self.inn` is a valid duckdb_appender; `begin_row` was called above.
71        let rc = unsafe { duckdb_appender_end_row(self.inn) };
72        result_from_duckdb_appender(rc, &mut self.inn)
73    }
74
75    /// Flushes all buffered rows to the database.
76    ///
77    /// # Errors
78    ///
79    /// Returns an error if the flush fails.
80    #[must_use = "save result should be checked"]
81    #[allow(dead_code)]
82    pub fn save(&mut self) -> Result<()> {
83        // SAFETY: `self.inn` is a valid duckdb_appender.
84        self.flush()
85    }
86
87    /// Flushes the appender's internal buffer.
88    ///
89    /// # Safety
90    ///
91    /// `self.inn` must be a valid, non-null `duckdb_appender`.
92    fn flush(&mut self) -> Result<()> {
93        // SAFETY: `self.inn` is a valid duckdb_appender (enforced by the caller).
94        let res = unsafe { duckdb_appender_flush(self.inn) };
95        result_from_duckdb_appender(res, &mut self.inn)
96    }
97}
98
99impl Drop for Appender {
100    fn drop(&mut self) {
101        if self.inn.is_null() {
102            return;
103        }
104        // [err-result-over-panic] — log on flush failure; never panic in Drop.
105        // SAFETY: `self.inn` is non-null (checked above); it is a valid duckdb_appender
106        // created in `new`. After close and destroy it is invalidated. The null guard
107        // above ensures this runs at most once.
108        if let Err(e) = self.flush() {
109            eprintln!("[better-duck] appender flush on drop failed: {e}");
110        }
111        // SAFETY: `self.inn` is a valid, non-null duckdb_appender (null guard above).
112        // Close and destroy are safe to call in sequence; after destroy the handle is
113        // invalid and will not be used again.
114        unsafe {
115            duckdb_appender_close(self.inn);
116            duckdb_appender_destroy(&mut self.inn);
117        }
118    }
119}

158    impl AppendAble for Row {
159        fn appender_append(
160            &mut self,
161            appender: duckdb_appender,
162        ) -> crate::error::Result<()> {
163            // SAFETY: `appender` is a valid duckdb_appender from `Appender::new`;
164            // we are inside a begin_row/end_row pair. The int32 and varchar values are
165            // valid for their respective columns.
166            unsafe {
167                duckdb_append_int32(appender, self.0);
168                let st = CString::new(self.1)
169                    .map_err(|e| DuckDBConversionError::ConversionError(e.to_string()))
170                    .unwrap();
171                duckdb_append_varchar(appender, st.as_ptr());
172            }
173            Ok(())
174        }
175        fn stmt_append(
176            &mut self,
177            idx: u64,
178            stmt: crate::ffi::duckdb_prepared_statement,
179        ) -> Result<()> {
180            // SAFETY: `stmt` is a valid prepared statement; `idx` is a 1-based parameter
181            // index within the statement's parameter count.
182            unsafe {
183                duckdb_bind_int32(stmt, idx, self.0);
184                let st = CString::new(self.1)
185                    .map_err(|e| DuckDBConversionError::ConversionError(e.to_string()))
186                    .unwrap();
187                duckdb_bind_varchar(stmt, idx + 1, st.as_ptr());
188            }
189            Ok(())
190        }
191    }

199    fn single_int(
200        con: &RawConnection,
201        sql: &str,
202        column: &str,
203    ) -> i32 {
204        let mut result = con.prepare(sql).unwrap().execute().unwrap();
205        let row = result.next().expect("expected one row").unwrap();
206        match row.get(column).unwrap() {
207            DuckValue::Int(value) => *value,
208            other => panic!("Expected Int for '{column}', got {other:?}"),
209        }
210    }

212    fn assert_row_exists(
213        con: &RawConnection,
214        table: &str,
215        id: i32,
216    ) {
217        let sql = format!("SELECT id FROM {table} WHERE id = {id}");
218        assert_eq!(single_int(con, &sql, "id"), id);
219    }

232    #[test]
233    fn test_appender_append_and_flush() {
234        let mut con = get_test_connection();
235
236        let _ = con.query("CREATE TABLE test_append (id INTEGER, name VARCHAR)").unwrap();
237
238        let mut appender = Appender::new(con.clone(), "test_append", "main").unwrap();
239        let mut row = Row(1, "Alice");
240        let mut row2 = Row(2, "Sara");
241        let mut row3 = Row(3, "Charlie");
242
243        appender.append(&mut row).unwrap();
244        appender.append(&mut row2).unwrap();
245        appender.append(&mut row3).unwrap();
246        appender.save().unwrap();
247
248        let mut stmt = con.prepare("SELECT id,name FROM test_append WHERE id=123").unwrap();
249        let mut rows = stmt.execute().unwrap();
250        assert!(rows.next().is_none(), "Row with id=123 should not exist");
251
252        let mut stmt = con.prepare("SELECT id,name FROM test_append").unwrap();
253        let rows = stmt.execute().unwrap();
254        for row in rows {
255            assert!(row.is_ok());
256            let row = row.unwrap();
257            let id = match row.get("id").unwrap() {
258                DuckValue::Int(id) => id,
259                other => panic!("Expected Int for 'id', got {:?}", other),
260            };
261            assert!([1, 2, 3].contains(id), "Row with id={} should exist", id);
262            let name = match row.get("name").unwrap() {
263                DuckValue::Text(name) => name.as_str(),
264                other => panic!("Expected Str for 'name', got {:?}", other),
265            };
266            match id {
267                1 => assert_eq!(name, "Alice"),
268                2 => assert_eq!(name, "Sara"),
269                3 => assert_eq!(name, "Charlie"),
270                _ => panic!("Unexpected row id: {}", id),
271            }
272        }
273    }
