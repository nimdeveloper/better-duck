26/// An in-progress scalar function overload, built up via its setters and then
27/// added to a [`ScalarFunctionSet`].
28pub(crate) struct ScalarFunction {
29    ptr: duckdb_scalar_function,
30}
31
32impl ScalarFunction {
33    pub(crate) fn new(name: &CStr) -> Self {
34        // SAFETY: always safe to call; returns a freshly allocated, empty function.
35        let ptr = unsafe { duckdb_create_scalar_function() };
36        // SAFETY: `ptr` was just allocated above and is non-null (DuckDB does not
37        // document a null return for this constructor).
38        unsafe { duckdb_scalar_function_set_name(ptr, name.as_ptr()) };
39        Self { ptr }
40    }
41
42    pub(crate) fn add_parameter(
43        &self,
44        ty: &LogicalType,
45    ) {
46        // SAFETY: `self.ptr` is valid; `ty.as_raw()` is a valid logical type owned
47        // by `ty` for the duration of this call. This function copies it.
48        unsafe { duckdb_scalar_function_add_parameter(self.ptr, ty.as_raw()) };
49    }
50
51    pub(crate) fn set_varargs(
52        &self,
53        ty: &LogicalType,
54    ) {
55        // SAFETY: `self.ptr` is valid; `ty.as_raw()` is a valid logical type owned
56        // by `ty` for the duration of this call. This function copies it.
57        unsafe { duckdb_scalar_function_set_varargs(self.ptr, ty.as_raw()) };
58    }
59
60    pub(crate) fn set_return_type(
61        &self,
62        ty: &LogicalType,
63    ) {
64        // SAFETY: `self.ptr` is valid; `ty.as_raw()` is a valid logical type owned
65        // by `ty` for the duration of this call. This function copies it.
66        unsafe { duckdb_scalar_function_set_return_type(self.ptr, ty.as_raw()) };
67    }
68
69    pub(crate) fn set_volatile(&self) {
70        // SAFETY: `self.ptr` is valid.
71        unsafe { duckdb_scalar_function_set_volatile(self.ptr) };
72    }
73
74    pub(crate) fn set_special_handling(&self) {
75        // SAFETY: `self.ptr` is valid.
76        unsafe { duckdb_scalar_function_set_special_handling(self.ptr) };
77    }
78
79    /// Sets the function's execution callback.
80    pub(crate) fn set_function(
81        &self,
82        f: duckdb_scalar_function_t,
83    ) {
84        // SAFETY: `self.ptr` is valid; `f`, if `Some`, is a valid
85        // `extern "C"` function pointer with the expected signature.
86        unsafe { duckdb_scalar_function_set_function(self.ptr, f) };
87    }
88
89    /// Stores `state`, retrievable inside the execution callback via
90    /// [`ScalarFunctionInfo::state`]. Freed automatically when DuckDB drops the
91    /// catalog entry.
92    pub(crate) fn set_extra_info<T: Send + Sync + 'static>(
93        &self,
94        state: T,
95    ) {
96        let ptr = Box::into_raw(Box::new(state)).cast::<c_void>();
97        // SAFETY: `ptr` was just created by `Box::into_raw::<T>` above;
98        // `drop_boxed::<T>` frees it with the matching type, exactly once, when
99        // DuckDB calls the delete callback.
100        unsafe { duckdb_scalar_function_set_extra_info(self.ptr, ptr, Some(drop_boxed::<T>)) };
101    }
102
103    pub(crate) fn as_raw(&self) -> duckdb_scalar_function {
104        self.ptr
105    }
106}
107
108impl Drop for ScalarFunction {
109    fn drop(&mut self) {
110        if !self.ptr.is_null() {
111            // SAFETY: `self.ptr` is a valid, non-null scalar function allocated by
112            // `duckdb_create_scalar_function` and not yet destroyed; destroyed
113            // exactly once here. Adding it to a set (`duckdb_add_scalar_function_to_set`)
114            // copies it, so destroying our own handle afterward is correct.
115            unsafe { duckdb_destroy_scalar_function(&mut self.ptr) };
116        }
117    }
118}

120/// A named collection of scalar function overloads, registered together.
121pub(crate) struct ScalarFunctionSet {
122    ptr: duckdb_scalar_function_set,
123}
124
125impl ScalarFunctionSet {
126    pub(crate) fn new(name: &CStr) -> Self {
127        // SAFETY: `name` is a valid, NUL-terminated C string for the duration of
128        // this call.
129        let ptr = unsafe { duckdb_create_scalar_function_set(name.as_ptr()) };
130        Self { ptr }
131    }
132
133    /// Adds `function` as a new overload.
134    ///
135    /// # Errors
136    ///
137    /// Returns an error if the overload conflicts with one already in the set.
138    pub(crate) fn add_function(
139        &self,
140        function: &ScalarFunction,
141    ) -> Result<()> {
142        // SAFETY: `self.ptr` and `function.as_raw()` are both valid; this
143        // function copies the scalar function into the set.
144        let rc = unsafe { duckdb_add_scalar_function_to_set(self.ptr, function.as_raw()) };
145        if rc != crate::ffi::DuckDBSuccess {
146            return Err(Error::DuckDBFailure(
147                crate::ffi::Error::new(rc),
148                Some(
149                    "failed to add overload to scalar function set (conflicting signature?)"
150                        .to_owned(),
151                ),
152            ));
153        }
154        Ok(())
155    }
156
157    /// Registers this set with `con`.
158    ///
159    /// # Errors
160    ///
161    /// Returns an error if registration fails, e.g. a name conflict.
162    pub(crate) fn register(
163        &self,
164        con: duckdb_connection,
165        name: &str,
166    ) -> Result<()> {
167        // SAFETY: `con` is a valid connection handle; `self.ptr` is a valid,
168        // non-empty function set (the caller adds overloads before calling this).
169        let rc = unsafe { duckdb_register_scalar_function_set(con, self.ptr) };
170        if rc != crate::ffi::DuckDBSuccess {
171            return Err(Error::DuckDBFailure(
172                crate::ffi::Error::new(rc),
173                Some(format!(
174                    "failed to register scalar function `{name}` (name conflict, or invalid signature)"
175                )),
176            ));
177        }
178        Ok(())
179    }
180}
181
182impl Drop for ScalarFunctionSet {
183    fn drop(&mut self) {
184        if !self.ptr.is_null() {
185            // SAFETY: `self.ptr` is a valid, non-null function set allocated by
186            // `duckdb_create_scalar_function_set` and not yet destroyed; destroyed
187            // exactly once here.
188            unsafe { duckdb_destroy_scalar_function_set(&mut self.ptr) };
189        }
190    }
191}

193/// The execution-time handle passed to [`super::VScalar::invoke`]'s trampoline,
194/// used to retrieve `State` and to report an error back to DuckDB.
195pub(crate) struct ScalarFunctionInfo {
196    ptr: duckdb_function_info,
197}
198
199impl ScalarFunctionInfo {
200    pub(crate) fn from(ptr: duckdb_function_info) -> Self {
201        Self { ptr }
202    }
203
204    /// Retrieves the state stored via [`ScalarFunction::set_extra_info`].
205    ///
206    /// # Safety
207    ///
208    /// `T` must be the same type that was passed to `set_extra_info` when this
209    /// function was registered.
210    pub(crate) unsafe fn state<T>(&self) -> &T {
211        // SAFETY: `self.ptr` is valid for the duration of the callback; the
212        // caller guarantees `T` matches the type stored at registration time,
213        // and that stored value outlives every invocation (DuckDB frees it only
214        // when the catalog entry itself is dropped).
215        let raw = unsafe { duckdb_scalar_function_get_extra_info(self.ptr) };
216        // SAFETY: `raw` was produced by `Box::into_raw::<T>` in `set_extra_info`
217        // and has not been freed (guaranteed by the caller per the above).
218        unsafe { &*raw.cast::<T>() }
219    }
220}
221
222impl CallbackErrorSink for ScalarFunctionInfo {
223    fn set_c_error(
224        &self,
225        error: &CStr,
226    ) {
227        // SAFETY: `self.ptr` is a valid function-info handle for the duration of
228        // the callback; `error` is a valid, NUL-terminated C string.
229        unsafe { duckdb_scalar_function_set_error(self.ptr, error.as_ptr()) };
230    }
