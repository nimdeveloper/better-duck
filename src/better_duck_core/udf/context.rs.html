1//! Ambient callback context for the `duck_projection!`/`duck_extra_info!`/
2//! `duck_state!` ergonomics macros.
3//!
4//! The `#[duckdb_table_function]`/`#[duckdb_scalar]` attribute macros call the
5//! user's plain function from *inside* their own generated `init`/`invoke`
6//! trampolines — a different Rust function from the user's body, with no
7//! parameter through which to reach `InitInfo`/the scalar `State`. These
8//! guards stash a type-erased pointer in a thread-local for the exact
9//! duration of that one call; the macros above read it back and clone the
10//! value out (hence the `Clone` bound on `T`), so the returned value is
11//! always independently owned — no reference to ambient context ever escapes,
12//! which sidesteps the soundness problems a bare "conjure a reference from a
13//! thread-local" API would otherwise have (nothing stops a caller from
14//! stashing such a reference somewhere that outlives the callback).
15//!
16//! Calling one of the macros outside its matching callback panics with a
17//! clear message. That panic is always contained by
18//! [`contain_callback`](super::callback::contain_callback), so it surfaces as
19//! a normal query error — never undefined behavior.
20
21use std::cell::Cell;
22use std::ffi::c_void;
23
24thread_local! {
25    static CURRENT_PROJECTION: Cell<*const Vec<usize>> = const { Cell::new(std::ptr::null()) };
26    static CURRENT_TABLE_EXTRA_INFO: Cell<*const c_void> = const { Cell::new(std::ptr::null()) };
27    static CURRENT_SCALAR_STATE: Cell<*const c_void> = const { Cell::new(std::ptr::null()) };
28}
29
30/// Sets the ambient projected-column-indices context for its lifetime,
31/// restoring whatever was there before on drop.
32pub struct ProjectionGuard(*const Vec<usize>);
33
34impl ProjectionGuard {
35    /// # Safety
36    ///
37    /// `indices` must remain valid (not moved, not dropped) for the entire
38    /// lifetime of the returned guard.
39    pub unsafe fn enter(indices: &Vec<usize>) -> Self {
40        let prev = CURRENT_PROJECTION.with(|c| c.replace(indices as *const Vec<usize>));
41        Self(prev)
42    }
43}
44
45impl Drop for ProjectionGuard {
46    fn drop(&mut self) {
47        CURRENT_PROJECTION.with(|c| c.set(self.0));
48    }
49}

51/// Sets the ambient table-function "extra info" context for its lifetime,
52/// restoring whatever was there before on drop.
53pub struct TableExtraInfoGuard(*const c_void);
54
55impl TableExtraInfoGuard {
56    /// # Safety
57    ///
58    /// `ptr`, if non-null, must remain valid for the entire lifetime of the
59    /// returned guard, and must point to a value of whatever type
60    /// `duck_extra_info!` is later instantiated with at the matching call site.
61    pub unsafe fn enter(ptr: *const c_void) -> Self {
62        let prev = CURRENT_TABLE_EXTRA_INFO.with(|c| c.replace(ptr));
63        Self(prev)
64    }
65}
66
67impl Drop for TableExtraInfoGuard {
68    fn drop(&mut self) {
69        CURRENT_TABLE_EXTRA_INFO.with(|c| c.set(self.0));
70    }
71}

95/// Reads back the current projected-column-indices context, cloning it out.
96///
97/// # Panics
98///
99/// Panics if called outside a table function's `init` callback (contained by
100/// `contain_callback`, so it surfaces as a normal query error).
101#[doc(hidden)]
102pub fn current_projection() -> Vec<usize> {
103    CURRENT_PROJECTION.with(|c| {
104        let ptr = c.get();
105        assert!(
106            !ptr.is_null(),
107            "duck_projection!() called outside a table function's init callback"
108        );
109        // SAFETY: `ptr` is non-null, so it was set by `ProjectionGuard::enter`
110        // and is still valid for the duration of the callback that guard
111        // wraps — which strictly contains this call, since the guard is only
112        // dropped after the user's plain fn (the only place this macro can
113        // syntactically appear) returns.
114        unsafe { (*ptr).clone() }
115    })
116}

118/// Reads back the current table-function "extra info" context, cloning it out.
119///
120/// # Panics
121///
122/// Panics if called outside a table function callback that was registered
123/// with extra info (contained by `contain_callback`).
124#[doc(hidden)]
125pub fn current_table_extra_info<T: Clone>() -> T {
126    CURRENT_TABLE_EXTRA_INFO.with(|c| {
127        let ptr = c.get();
128        assert!(
129            !ptr.is_null(),
130            "duck_extra_info!() called outside a table function callback registered with extra info \
131             (see `Connection::register_table_function_with_extra_info`)"
132        );
133        // SAFETY: see `current_projection` — same guard-lifetime argument.
134        // The caller is responsible for requesting the same `T` that was
135        // stored at registration time; there is no runtime type check.
136        unsafe { (*ptr.cast::<T>()).clone() }
137    })
138}

140/// Reads back the current scalar function `State` context, cloning it out.
141///
142/// # Panics
143///
144/// Panics if called outside a `#[duckdb_scalar]`-generated callback.
145#[doc(hidden)]
146pub fn current_scalar_state<T: Clone>() -> T {
147    CURRENT_SCALAR_STATE.with(|c| {
148        let ptr = c.get();
149        assert!(!ptr.is_null(), "duck_state!() called outside a scalar function callback");
150        // SAFETY: see `current_projection` — same guard-lifetime argument.
151        // The caller is responsible for requesting the same `T` that the
152        // enclosing `#[duckdb_scalar]` function declared as its state; there
153        // is no runtime type check.
154        unsafe { (*ptr.cast::<T>()).clone() }
155    })
156}

158#[cfg(test)]
159mod tests {
160    use super::*;
161
162    #[test]
163    #[should_panic(expected = "duck_projection!() called outside")]
164    fn current_projection_panics_without_a_guard() {
165        let _ = current_projection();
166    }
167
168    #[test]
169    fn projection_guard_round_trips_and_restores_previous_value() {
170        let outer = vec![1usize, 2, 3];
171        // SAFETY: `outer` outlives both guards below.
172        let outer_guard = unsafe { ProjectionGuard::enter(&outer) };
173        assert_eq!(current_projection(), outer);
174
175        {
176            let inner = vec![9usize];
177            // SAFETY: `inner` outlives this guard.
178            let _inner_guard = unsafe { ProjectionGuard::enter(&inner) };
179            assert_eq!(current_projection(), inner);
180        }
181
182        assert_eq!(
183            current_projection(),
184            outer,
185            "dropping the inner guard must restore the outer one"
186        );
187        drop(outer_guard);
188    }
189
190    #[test]
191    #[should_panic(expected = "duck_extra_info!() called outside")]
192    fn current_table_extra_info_panics_without_a_guard() {
193        let _: i32 = current_table_extra_info::<i32>();
194    }
195
196    #[test]
197    fn table_extra_info_guard_round_trips() {
198        let value = 42i32;
199        // SAFETY: `value` outlives the guard.
200        let _guard = unsafe {
201            TableExtraInfoGuard::enter((&value as *const i32).cast::<std::ffi::c_void>())
202        };
203        assert_eq!(current_table_extra_info::<i32>(), 42);
204    }
205
206    #[test]
207    #[should_panic(expected = "duck_state!() called outside")]
208    fn current_scalar_state_panics_without_a_guard() {
209        let _: i32 = current_scalar_state::<i32>();
210    }
211
212    #[test]
213    fn scalar_state_guard_round_trips() {
214        let value = 7i32;
215        // SAFETY: `value` outlives the guard.
216        let _guard =
217            unsafe { ScalarStateGuard::enter((&value as *const i32).cast::<std::ffi::c_void>()) };
218        assert_eq!(current_scalar_state::<i32>(), 7);
219    }
