1//! Function-signature validation and type introspection shared by
2//! `#[duckdb_scalar]` and `#[duckdb_table_function]`.
3
4use syn::{FnArg, GenericArgument, ItemFn, Pat, PathArguments, Safety, Type};
5
6/// One parameter of a user-annotated function.
7pub(crate) struct Param {
8    /// Unused by `#[duckdb_scalar]` (parameters are matched positionally);
9    /// used by `#[duckdb_table_function]` to match `named_params(...)` entries
10    /// against actual parameter names.
11    pub(crate) ident: syn::Ident,
12    /// The parameter's full declared type, `Option<T>` wrapper included if
13    /// present — both `better_duck_core`'s `ScalarArg` and `DuckLogicalType`
14    /// traits have a blanket `Option<T>` impl, so the wrapper never needs to be
15    /// stripped for codegen purposes.
16    pub(crate) ty: Type,
17    /// Whether `ty` is syntactically `Option<...>` — used only to decide
18    /// whether this parameter is exempt from the pre-call null-skip guard and
19    /// whether the function needs `special_handling()`.
20    pub(crate) is_option: bool,
21}
22
23/// Rejects `async`, `unsafe`, non-Rust ABI, generics/`where`, and a `self`
24/// receiver — none of which make sense for a DuckDB UDF, which is registered
25/// as a bare, monomorphic function pointer.
26pub(crate) fn validate_shape(item: &ItemFn) -> syn::Result<()> {
27    if let Some(asyncness) = &item.sig.asyncness {
28        return Err(syn::Error::new_spanned(asyncness, "DuckDB UDFs cannot be `async`"));
29    }
30    if let Safety::Unsafe(unsafety) = &item.sig.safety {
31        return Err(syn::Error::new_spanned(unsafety, "DuckDB UDFs must be a safe Rust `fn`"));
32    }
33    if let Some(abi) = &item.sig.abi {
34        return Err(syn::Error::new_spanned(abi, "DuckDB UDFs must not specify an ABI"));
35    }
36    if !item.sig.generics.params.is_empty() || item.sig.generics.where_clause.is_some() {
37        return Err(syn::Error::new_spanned(&item.sig.generics, "DuckDB UDFs must not be generic"));
38    }
39    for arg in &item.sig.inputs {
40        if let FnArg::Receiver(recv) = arg {
41            return Err(syn::Error::new_spanned(
42                recv,
43                "DuckDB UDFs must be a free function, not a method",
44            ));
45        }
46    }
47    Ok(())
48}

50/// Extracts the parameter list, in declaration order.
51///
52/// Each parameter's pattern must be a simple identifier (no destructuring) —
53/// the generated code needs one name per parameter to read into.
54pub(crate) fn extract_params(item: &ItemFn) -> syn::Result<Vec<Param>> {
55    item.sig
56        .inputs
57        .iter()
58        .map(|arg| {
59            let FnArg::Typed(pat_ty) = arg else {
60                unreachable!("receivers are rejected by validate_shape before this runs")
61            };
62            let ident = match &*pat_ty.pat {
63                Pat::Ident(pi) => pi.ident.clone(),
64                other => {
65                    return Err(syn::Error::new_spanned(
66                        other,
67                        "DuckDB UDF parameters must be a simple identifier, not a pattern",
68                    ))
69                },
70            };
71            let ty = (*pat_ty.ty).clone();
72            let is_option = is_option_type(&ty);
73            Ok(Param { ident, ty, is_option })
74        })
75        .collect()
76}

78/// Returns `true` if `ty` is syntactically `Option<...>`.
79pub(crate) fn is_option_type(ty: &Type) -> bool {
80    let Type::Path(type_path) = ty else { return false };
81    type_path.path.segments.last().is_some_and(|seg| seg.ident == "Option")
82}

84/// If `ty` is syntactically `Result<T, E>`, returns `(true, Some(E), T)`;
85/// otherwise `(false, None, ty)` unchanged.
86pub(crate) fn unwrap_result(ty: &Type) -> (bool, Option<Type>, Type) {
87    if let Type::Path(type_path) = ty {
88        if let Some(seg) = type_path.path.segments.last() {
89            if seg.ident == "Result" {
90                if let PathArguments::AngleBracketed(args) = &seg.arguments {
91                    let mut iter = args.args.iter();
92                    if let (Some(GenericArgument::Type(ok)), Some(GenericArgument::Type(err))) =
93                        (iter.next(), iter.next())
94                    {
95                        return (true, Some(err.clone()), ok.clone());
96                    }
97                }
98            }
99        }
100    }
101    (false, None, ty.clone())
102}
